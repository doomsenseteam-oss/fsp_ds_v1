    local SuppressedSounds = {}

    local function SuppressAudio(snd)
        if not snd or not snd:IsA("Sound") or snd.Name:find("Custom_") then return end
        local nameLow = snd.Name:lower()
        if nameLow:find("lobby") then return end

        if not SuppressedSounds[snd] then
            local eq = snd:FindFirstChild("FPlus_SilentFilter")
            if not eq then
                eq = Instance.new("EqualizerSoundEffect")
                eq.Name = "FPlus_SilentFilter"
                eq.LowGain = -80
                eq.MidGain = -80
                eq.HighGain = -80
                eq.Priority = 1000
                eq.Parent = snd
            end

            SuppressedSounds[snd] = {
                OriginalGroup = snd.SoundGroup,
                Filter = eq
            }
            snd.SoundGroup = MuteBus
        else
            if snd.SoundGroup ~= MuteBus then
                snd.SoundGroup = MuteBus
            end
        end
    end

    local function RestoreGameThemes()
        for snd, data in pairs(SuppressedSounds) do
            if snd and snd.Parent then
                pcall(function() snd.SoundGroup = data.OriginalGroup end)
                if data.Filter and data.Filter.Parent then
                    data.Filter:Destroy()
                end
            end
        end
        table.clear(SuppressedSounds)
    end

    local function SafeList(dir)
        local results = {}
        if not listfiles then return results end
        for _, path in ipairs({ dir, dir:gsub("/", "\\"), dir .. "/", dir:gsub("/", "\\") .. "\\" }) do
            local s, files = pcall(listfiles, path)
            if s and type(files) == "table" and #files > 0 then
                for _, f in ipairs(files) do
                    local fullPath = tostring(f):gsub("\\", "/")
                    if not fullPath:find("/") then
                        fullPath = dir:gsub("\\", "/") .. "/" .. fullPath
                    end
                    results[#results + 1] = fullPath
                end
                break
            end
        end
        return results
    end

    local function ParseChaseTiming(txtPath)
        local timings = {}
        if isfile and readfile then
            local exists = false
            pcall(function() exists = isfile(txtPath) or isfile(txtPath:gsub("/", "\\")) end)
            if exists then
                local content = ""
                pcall(function() content = readfile(isfile(txtPath) and txtPath or txtPath:gsub("/", "\\")) end)
                for line in content:gmatch("[^\r\n]+") do
                    local key, sStart, sEnd = line:match("([%w_%-]+)%s*:%s*([%d%.]+)%s*-%s*([%d%.]+)")
                    if key and sStart and sEnd then
                        local k = key:lower():gsub("%s+", "")
                        local startT, endT = tonumber(sStart), tonumber(sEnd)
                        if startT and endT and endT > startT then
                            if k:find("lay1%-2") or k:find("lay1_2") or k:find("lay1to2") or k:find("layer1%-2") then
                                timings.Trans1_2 = { Start = startT, End = endT }
                            elseif k:find("lay2%-3") or k:find("lay2_3") or k:find("lay2to3") or k:find("layer2%-3") then
                                timings.Trans2_3 = { Start = startT, End = endT }
                            elseif k:find("lay3%-chase") or k:find("lay3_chase") or k:find("lay3tochase") or k:find("layer3%-chase") then
                                timings.Trans3_Chase = { Start = startT, End = endT }
                            elseif k:find("lay1") or k:find("layer1") then
                                timings.Lay1 = { Start = startT, End = endT }
                            elseif k:find("lay2") or k:find("layer2") then
                                timings.Lay2 = { Start = startT, End = endT }
                            elseif k:find("lay3") or k:find("layer3") then
                                timings.Lay3 = { Start = startT, End = endT }
                            elseif k:find("chase") then
                                timings.Chase = { Start = startT, End = endT }
                            end
                        end
                    end
                end
            end
        end
        return timings
    end

    local function ScanAllTracks()
        table.clear(TrackRegistry.Survivor)
        table.clear(TrackRegistry.Killer)
        table.clear(TrackRegistry.Chase)

        local survList = {"None", "Off"}
        local killList = {"None", "Off"}
        local chaseList = {"None", "Off"}
        local seenSurv, seenKill, seenChase = {}, {}, {}

        local folders = { SURV_DIR, KILL_DIR, LMS_DIR, CHASE_DIR, MUSIC_DIR, BASE_ROOT }

        local function CheckFile(fullPath)
            local norm = fullPath:gsub("\\", "/")
            local filename = norm:match("[^/]+$")
            if not filename then return end

            local ext = filename:match("%.([^%.]+)$")
            if ext then
                ext = ext:lower()
                if ext == "mp3" or ext == "ogg" or ext == "wav" then
                    local cleanName = filename:sub(1, #filename - #ext - 1)
                    local lowPath = norm:lower()
                    local lowName = cleanName:lower()

                    if lowPath:find("chase") then
                        if not seenChase[lowName] then
                            seenChase[lowName] = true
                            local txtPath = norm:sub(1, #norm - #ext) .. "txt"
                            TrackRegistry.Chase[cleanName] = {
                                AudioPath = norm,
                                Timings = ParseChaseTiming(txtPath)
                            }
                            table.insert(chaseList, cleanName)
                        end
                    end

                    local isKiller = lowPath:find("/killer") or lowPath:find("\\killer")
                    local isChase  = lowPath:find("chase")
                    if not isKiller and not isChase then
                        if not seenSurv[lowName] then
                            seenSurv[lowName] = true
                            TrackRegistry.Survivor[cleanName] = norm
                            table.insert(survList, cleanName)
                        end
                    end

                    local isSurv = lowPath:find("/survivor") or lowPath:find("\\survivor")
                    if not isSurv and not isChase then
                        if not seenKill[lowName] then
                            seenKill[lowName] = true
                            TrackRegistry.Killer[cleanName] = norm
                            table.insert(killList, cleanName)
                        end
                    end
                end
            end
        end

        for _, fld in ipairs(folders) do
            for _, f in ipairs(SafeList(fld)) do
                CheckFile(f)
            end
        end

        local function SortOptions(tbl)
            table.sort(tbl, function(a, b)
                if a == "None" then return true elseif b == "None" then return false end
                if a == "Off" then return true elseif b == "Off" then return false end
                return a:lower() < b:lower()
            end)
        end
        SortOptions(survList)
        SortOptions(killList)
        SortOptions(chaseList)

        return table.concat(survList, "|"), table.concat(killList, "|"), table.concat(chaseList, "|")
    end

    local function ResolveTrack(role, selectedTrack)
        if not selectedTrack or selectedTrack == "None" or selectedTrack == "Off" then return nil end
        local finalPath = nil
        if role == "Chase" then
            finalPath = TrackRegistry.Chase[selectedTrack] and TrackRegistry.Chase[selectedTrack].AudioPath
        else
            finalPath = TrackRegistry[role] and TrackRegistry[role][selectedTrack]
        end

        if not finalPath then
            ScanAllTracks()
            if role == "Chase" then
                finalPath = TrackRegistry.Chase[selectedTrack] and TrackRegistry.Chase[selectedTrack].AudioPath
            else
                finalPath = TrackRegistry[role] and TrackRegistry[role][selectedTrack]
            end
        end

        if not finalPath and isfile then
            local checkDirs = { SURV_DIR, KILL_DIR, LMS_DIR, CHASE_DIR, MUSIC_DIR, BASE_ROOT }
            for _, d in ipairs(checkDirs) do
                for _, ext in ipairs({ ".mp3", ".ogg", ".wav" }) do
                    local directPath = d .. "/" .. selectedTrack .. ext
                    local exists = false
                    pcall(function() exists = isfile(directPath) or isfile(directPath:gsub("/", "\\")) end)
                    if exists then
                        finalPath = directPath
                        break
                    end
                end
                if finalPath then break end
            end
        end

        if finalPath and (getcustomasset or getsynasset) then
            local func = getcustomasset or getsynasset
            local s, asset = pcall(func, finalPath)
            if s and asset then return asset end
            local s2, asset2 = pcall(func, finalPath:gsub("/", "\\"))
            if s2 and asset2 then return asset2 end
        end
        return nil
    end

    local function DetectRole(model)
        if not model or not model:IsA("Model") then return "Survivor" end
        local killersFolder = workspace.Players:FindFirstChild("Killers")
        local survFolder = workspace.Players:FindFirstChild("Survivors")

        if killersFolder and model:IsDescendantOf(killersFolder) then return "Killer" end
        if survFolder and model:IsDescendantOf(survFolder) then return "Survivor" end

        local hum = model:FindFirstChildOfClass("Humanoid")
        if hum and hum.MaxHealth >= 500 then return "Killer" end

        local clean = model.Name:lower():gsub("[%s_%-]+", "")
        for k, _ in pairs(KILLER_KEYS) do
            if clean:find(k) then return "Killer" end
        end
        return "Survivor"
    end

    local function ResolveEntity(model, fallbackRole)
        local rawNames = {}
        if model and model:IsA("Model") then
            local plr = Players:GetPlayerFromCharacter(model)
            if plr then table.insert(rawNames, plr.DisplayName) end
            table.insert(rawNames, model:GetAttribute("Character"))
            table.insert(rawNames, model:GetAttribute("CharName"))
            table.insert(rawNames, model:GetAttribute("DisplayName"))
            table.insert(rawNames, model:GetAttribute("Actor"))
            table.insert(rawNames, model.Name)
        end

        for _, raw in ipairs(rawNames) do
            if raw and typeof(raw) == "string" and raw ~= "" then
                local clean = raw:lower():gsub("[%s_%-]+", "")
                for key, data in pairs(CHAR_DATA) do
                    local keyClean = key:lower():gsub("[%s_%-]+", "")
                    if clean == keyClean or clean:find("^" .. keyClean) or clean:find(keyClean .. "$") then
                        return "rbxthumb://type=Asset&id=" .. data.Asset .. "&w=420&h=420", data.Name
                    end
                end
            end
        end

        if fallbackRole == "Killer" then
            return "rbxthumb://type=Asset&id=89705807157114&w=420&h=420", (model and model.Name or "Killer")
        else
            return "rbxthumb://type=Asset&id=131391655751675&w=420&h=420", (model and model.Name or "Survivor")
        end
    end

    local function StopLMS(instant)
        LMS_State.IsTriggered = false
        LMS_State.LMSActive = false
        LMS_State.VisualPlayed = false
        LMS_State.StopAt = nil
        LMS_State.CurrentTrackName = nil

        if FX_Gui then
            pcall(function() FX_Gui:ClearAllChildren() end)
        end

        if LMS_State.CurrentSound then
            local snd = LMS_State.CurrentSound
            LMS_State.CurrentSound = nil
            if instant then
                snd:Stop()
                snd:Destroy()
            else
                local tween = TweenService:Create(snd, TweenInfo.new(0.2, Enum.EasingStyle.Sine, Enum.EasingDirection.Out), { Volume = 0 })
                tween:Play()
                tween.Completed:Once(function()
                    snd:Stop()
                    snd:Destroy()
                end)
            end
        end

        local isChaseRunning = GetValue("Chase_Active") and (GetValue("Chase_Track") ~= "None" and GetValue("Chase_Track") ~= "Off")
        if not isChaseRunning then
            RestoreGameThemes()
        end
    end

    end

    local function CheckTrueLMS()
        if not GetValue("LMS_Active") then
            if LMS_State.IsTriggered or LMS_State.CurrentSound then StopLMS(true) end
            return
        end

        local inLobby = IsInLobby()
        local allowLobbyAudio = (GetValue("LMS_LobbyAudio") == true)

        if inLobby and not allowLobbyAudio then
            if LMS_State.IsTriggered or LMS_State.CurrentSound then StopLMS(true) end
            return
        end

        local themes = workspace:FindFirstChild("Themes")
        local lastSurv = themes and themes:FindFirstChild("LastSurvivor")
        local isNativeLMS = lastSurv and lastSurv:IsA("Sound") and (lastSurv.IsPlaying or lastSurv.Playing or lastSurv.TimePosition > 0)

        local survivorsFolder = workspace.Players:FindFirstChild("Survivors")
        local spectatingFolder = workspace.Players:FindFirstChild("Spectating")
        local killersFolder = workspace.Players:FindFirstChild("Killers")

        if not survivorsFolder or not killersFolder then
            if LMS_State.IsTriggered then StopLMS(true) end
            return
        end

        local activeSurvivors = 0
        local totalSurvivors = 0
        local myCharacter = LocalPlayer.Character
        local myHum = myCharacter and myCharacter:FindFirstChildOfClass("Humanoid")
        local isAlive = myCharacter and (myCharacter:GetAttribute("Dead") ~= true) and (myHum and myHum.Health > 0)

        local isLocalSurvivor = isAlive and myCharacter:IsDescendantOf(survivorsFolder)
        local isLocalKiller = isAlive and myCharacter:IsDescendantOf(killersFolder)

        for _, char in ipairs(survivorsFolder:GetChildren()) do
            totalSurvivors = totalSurvivors + 1
            local hum = char:FindFirstChildOfClass("Humanoid")
            local isDead = (char:GetAttribute("Dead") == true) or (hum and hum.Health <= 0)
            local isSpectating = spectatingFolder and char:IsDescendantOf(spectatingFolder)

            if not isDead and not isSpectating then
                activeSurvivors = activeSurvivors + 1
            end
        end

        local isLMS = false
        if isNativeLMS then
            isLMS = true
        elseif activeSurvivors == 1 and totalSurvivors > 1 and not inLobby then
            isLMS = true
        end

        if inLobby and not allowLobbyAudio then
            isLMS = false
        end

        if isLMS then
            LMS_State.StopAt = nil

            local role = "Survivor"
            local selectedTrack = "None"

            if inLobby and allowLobbyAudio then
                role = "Survivor"
                selectedTrack = GetValue("LMS_SurvTrack") or "None"
            elseif isLocalSurvivor then
                role = "Survivor"
                selectedTrack = GetValue("LMS_SurvTrack") or "None"
            elseif isLocalKiller then
                role = "Killer"
                selectedTrack = GetValue("LMS_KillerTrack") or "None"
            else
                local cam = workspace.CurrentCamera
                local subject = cam and cam.CameraSubject
                local spectatedChar = subject and (subject:IsA("Humanoid") and subject.Parent or subject:IsA("Model") and subject)

                if spectatedChar and spectatedChar:IsDescendantOf(killersFolder) then
                    role = "Killer"
                    selectedTrack = GetValue("LMS_KillerTrack") or "None"
                else
                    role = "Survivor"
                    selectedTrack = GetValue("LMS_SurvTrack") or "None"
                end
            end

            -- [ Off Mode ]
            if selectedTrack == "Off" then
                if LMS_State.CurrentSound then
                    StopLMS(false)
                end
                local isChaseRunning = GetValue("Chase_Active") and (GetValue("Chase_Track") ~= "None" and GetValue("Chase_Track") ~= "Off")
                if not isChaseRunning then
                    RestoreGameThemes()
                end
                return
            end

            -- MuteBus
            if themes then
                for _, th in ipairs(themes:GetChildren()) do
                    if th:IsA("Sound") and not th.Name:find("Custom_") then
                        SuppressAudio(th)
                    end
                end
            end

            -- Hotswap
            if LMS_State.IsTriggered and LMS_State.CurrentTrackName ~= selectedTrack then
                LMS_State.CurrentTrackName = selectedTrack
                if selectedTrack == "None" then
                    if LMS_State.CurrentSound then StopLMS(false) end
                else
                    local assetId = ResolveTrack(role, selectedTrack)
                    if assetId then PlayLocalLMS(assetId, selectedTrack) end
                end
            end

            if not LMS_State.IsTriggered then
                LMS_State.IsTriggered = true
                LMS_State.LMSActive = true
                local delaySec = tonumber(GetValue("LMS_StartDelay")) or 0

                task.delay(delaySec, function()
                    if not LMS_State.IsTriggered or not GetValue("LMS_Active") then return end
                    if IsInLobby() and not (GetValue("LMS_LobbyAudio") == true) then return end

                    TriggerLMSVisualEffect()

                    if selectedTrack ~= "None" and selectedTrack ~= "Off" then
                        local assetId = ResolveTrack(role, selectedTrack)
                        if assetId then PlayLocalLMS(assetId, selectedTrack) end
                    end
                end)
            else
                if LMS_State.CurrentSound and not LMS_State.CurrentSound.IsPlaying then
                    LMS_State.CurrentSound:Play()
                end
            end
        else
            if LMS_State.IsTriggered then
                if not LMS_State.StopAt then
                    LMS_State.StopAt = os.clock() + 2.0
                elseif os.clock() >= LMS_State.StopAt then
                    StopLMS(true)
                end
            end
        end
    end

    local function IsInCutscene()
        local themes = workspace:FindFirstChild("Themes")
        if themes then
            for _, snd in ipairs(themes:GetChildren()) do
                if snd:IsA("Sound") and (snd.IsPlaying or snd.Playing) then
                    local n = snd.Name:lower()
                    if n:find("victory") or n:find("intro") or n:find("execution") then
                        return true
                    end
                end
            end
        end

        local char = LocalPlayer.Character
        if char then
            if char:GetAttribute("InCutscene") == true
                or char:GetAttribute("Executing") == true
                or char:GetAttribute("InExecution") == true
                or char:GetAttribute("IntroActive") == true then
                return true
            end

            local hum = char:FindFirstChildOfClass("Humanoid")
            local anim = hum and hum:FindFirstChildOfClass("Animator")
            if anim then
                for _, track in ipairs(anim:GetPlayingAnimationTracks()) do
                    local tName = (track.Name or ""):lower()
                    if tName:find("execution") or tName:find("intro") or tName:find("victory") then
                        return true
                    end
                end
            end
        end

        local rTimer = ReplicatedStorage:FindFirstChild("RoundTimer")
        if rTimer and (rTimer.Value == "Round begins in:" or rTimer.Value == "Waiting for more players.") then
            return true
        end

        return false
    end

    local function LoadTrackConfig(trackName)
        if not isfile or not readfile or not trackName or trackName == "None" or trackName == "Off" then return end
        local cfgPath = GetTrackConfigPath(trackName)
        if not cfgPath then return end

        local norm = cfgPath:gsub("/", "\\")
        local exists = false
        pcall(function() exists = isfile(cfgPath) or isfile(norm) end)

        if not exists then
            SaveTrackConfig(trackName)
            return
        end

        local rawContent = ""
        pcall(function()
            rawContent = readfile(isfile(cfgPath) and cfgPath or norm)
        end)

        local s, data = pcall(function() return HttpService:JSONDecode(rawContent) end)
        if s and type(data) == "table" then
            IsConfigLoading = true
            local function ApplyVal(key, val)
                local inst = GetValue(key, true)
                if inst and val ~= nil then
                    inst.Value = val
                end
            end

            ApplyVal("Chase_Volume", data.Volume)
            ApplyVal("Chase_FadeTime", data.FadeTime)
            ApplyVal("Chase_StartDelay", data.StartDelay)
            ApplyVal("Chase_RangeMultiplier", data.RangeMultiplier)
            ApplyVal("Chase_OverlapBuffer", data.OverlapBuffer)
            ApplyVal("Chase_BonusExpansion", data.BonusExpansion)
            ApplyVal("Chase_DistChase", data.DistChase)
            ApplyVal("Chase_DistLay3", data.DistLay3)
            ApplyVal("Chase_DistLay2", data.DistLay2)
            ApplyVal("Chase_DistLay1", data.DistLay1)
            ApplyVal("Chase_PlaybackSpeed", data.PlaybackSpeed)
            task.delay(0.1, function() IsConfigLoading = false end)
        end
    end

    local function SilenceThemesForCustomChase()
        if not IsLocalKiller() or IsInLobby() then return end
        local themes = workspace:FindFirstChild("Themes")
        if not themes then return end

        for _, snd in ipairs(themes:GetChildren()) do
            if snd:IsA("Sound") and not snd.Name:find("Custom_") then
                local n = snd.Name:lower()
                local isProtected = n:find("lobby") or n:find("execution") or n:find("intro")
                    or n:find("victory") or n:find("kill") or n:find("scare")
                if not isProtected then
                    SuppressAudio(snd)
                end
            end
        end
    end

    local function StopChaseAudio(instant)
        Chase_State.CurrentLayer = 0
        Chase_State.CurrentTierName = nil
        Chase_State.IsTransitioning = false
        Chase_State.ZeroSince = nil
        Chase_State.LastProcessedLayer = -1
        Chase_State.LastProcessedTrack = nil

        if Chase_State.LoopConnA then Chase_State.LoopConnA:Disconnect(); Chase_State.LoopConnA = nil end
        if Chase_State.LoopConnB then Chase_State.LoopConnB:Disconnect(); Chase_State.LoopConnB = nil end

        local fadeTime = instant and 0 or (tonumber(GetValue("Chase_FadeTime")) or 0.6)

        local function KillVoice(v)
            if not v then return end
            if instant or fadeTime <= 0.05 then
                v:Stop()
                v:Destroy()
            else
                local tw = TweenService:Create(v, TweenInfo.new(fadeTime, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), { Volume = 0 })
                tw:Play()
                tw.Completed:Once(function()
                    v:Stop()
                    v:Destroy()
                end)
                Debris:AddItem(v, fadeTime + 0.2)
            end
        end

        KillVoice(Chase_State.VoiceA); Chase_State.VoiceA = nil
        KillVoice(Chase_State.VoiceB); Chase_State.VoiceB = nil
        Chase_State.ActiveAsset = nil

        if Chase_State.ThemesSilenced then
            Chase_State.ThemesSilenced = false
            if not (LMS_State and LMS_State.LMSActive) then
                RestoreGameThemes()
            end
        end
    end

    -- [ Dual-Voice Crossfader ]
    local function PlayChaseSegment(assetId, tierName, segment, isTransition, nextTierName, nextSegment)
        local baseVol = tonumber(GetValue("Chase_Volume")) or 1.0
        local pitch = tonumber(GetValue("Chase_PlaybackSpeed")) or 1.0
        local fadeTime = math.clamp(tonumber(GetValue("Chase_FadeTime")) or 0.6, 0.1, 3.0)
        local distFactor = math.clamp(1 - (Chase_State.ClosestDistance / 100), 0, 1)
        local targetVol = baseVol * (0.8 + 0.25 * distFactor)

        local currentVoice = (Chase_State.ActiveVoice == "A") and Chase_State.VoiceA or Chase_State.VoiceB
        local targetVoiceSlot = (Chase_State.ActiveVoice == "A") and "B" or "A"

        local newVoice = Instance.new("Sound")
        newVoice.Name = "Custom_Chase_Voice_" .. targetVoiceSlot
        newVoice.SoundId = assetId
        newVoice.Volume = 0
        newVoice.PlaybackSpeed = pitch
        newVoice.Parent = SoundService
        newVoice:Play()

        if targetVoiceSlot == "B" then
            Chase_State.VoiceB = newVoice
            if Chase_State.LoopConnB then Chase_State.LoopConnB:Disconnect(); Chase_State.LoopConnB = nil end
        else
            Chase_State.VoiceA = newVoice
            if Chase_State.LoopConnA then Chase_State.LoopConnA:Disconnect(); Chase_State.LoopConnA = nil end
        end

        Chase_State.ActiveVoice = targetVoiceSlot
        Chase_State.ActiveAsset = assetId

        if currentVoice and currentVoice.IsPlaying then
            local fadeOut = TweenService:Create(currentVoice, TweenInfo.new(fadeTime, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), { Volume = 0 })
            fadeOut:Play()
            fadeOut.Completed:Once(function()
                currentVoice:Stop()
                currentVoice:Destroy()
            end)
            Debris:AddItem(currentVoice, fadeTime + 0.2)
        end

        local fadeIn = TweenService:Create(newVoice, TweenInfo.new(fadeTime, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), { Volume = targetVol })
        fadeIn:Play()

        local function SetupLoop()
            if not newVoice or not newVoice.Parent then return end
            if segment and segment.Start and segment.End then
                newVoice.Looped = false
                newVoice.TimePosition = segment.Start

                if isTransition then
                    Chase_State.IsTransitioning = true
                    Chase_State.CurrentTierName = tierName
                    local conn
                    conn = RunService.Heartbeat:Connect(function()
                        if newVoice and newVoice.IsPlaying then
                            if newVoice.TimePosition >= segment.End then
                                Chase_State.IsTransitioning = false
                                if conn then conn:Disconnect() end
                                if nextTierName and nextSegment then
                                    PlayChaseSegment(assetId, nextTierName, nextSegment, false, nil, nil)
                                end
                            end
                        else
                            if conn then conn:Disconnect() end
                        end
                    end)
                    if targetVoiceSlot == "B" then Chase_State.LoopConnB = conn else Chase_State.LoopConnA = conn end
                else
                    Chase_State.IsTransitioning = false
                    Chase_State.CurrentTierName = tierName
                    local conn
                    conn = RunService.Heartbeat:Connect(function()
                        if newVoice and newVoice.IsPlaying then
                            if newVoice.TimePosition >= segment.End then
                                newVoice.TimePosition = segment.Start
                            end
                        else
                            if conn then conn:Disconnect() end
                        end
                    end)
                    if targetVoiceSlot == "B" then Chase_State.LoopConnB = conn else Chase_State.LoopConnA = conn end
                end
            else
                Chase_State.IsTransitioning = false
                newVoice.Looped = true
                Chase_State.CurrentTierName = tierName
            end
        end

        if not newVoice.IsLoaded then
            newVoice.Loaded:Once(SetupLoop)
        else
            SetupLoop()
        end
    end

    local function ResolveGameActiveLayer()
        if IsInLobby() then return 0 end
        if not IsLocalKiller() then return 0 end
        if IsInCutscene() then return 0 end

        local myChar = LocalPlayer.Character
        if not myChar or myChar:GetAttribute("Dead") == true then return 0 end

        local myRoot = myChar:FindFirstChild("HumanoidRootPart") or myChar.PrimaryPart
        local survFolder = workspace.Players:FindFirstChild("Survivors")
        if not myRoot or not survFolder then return 0 end

        local survDistances = {}
        local minD = 9999
        for _, sChar in ipairs(survFolder:GetChildren()) do
            local sHum = sChar:FindFirstChildOfClass("Humanoid")
            local sRoot = sChar:FindFirstChild("HumanoidRootPart") or sChar.PrimaryPart
            if sRoot and (not sHum or sHum.Health > 0) and sChar:GetAttribute("Dead") ~= true then
                local d = (sRoot.Position - myRoot.Position).Magnitude
                table.insert(survDistances, d)
                if d < minD then
                    minD = d
                end
            end
        end
        Chase_State.ClosestDistance = minD

        local function CountInZone(radius)
            local c = 0
            for _, d in ipairs(survDistances) do
                if d <= radius then c = c + 1 end
            end
            return c
        end

        local mult    = math.max(tonumber(GetValue("Chase_RangeMultiplier")) or 1.0, 0.1)
        local buffer  = math.clamp(tonumber(GetValue("Chase_OverlapBuffer")) or 3.57, 0, 15)
        local bonus   = math.max(tonumber(GetValue("Chase_BonusExpansion")) or 25.0, 0)

        local rChaseBase = (tonumber(GetValue("Chase_DistChase")) or 22) * mult
        local rLay3Base  = (tonumber(GetValue("Chase_DistLay3")) or 40) * mult
        local rLay2Base  = (tonumber(GetValue("Chase_DistLay2")) or 60) * mult
        local rLay1Base  = (tonumber(GetValue("Chase_DistLay1")) or 85) * mult

        local rTotalSize = rLay1Base

        local rChaseAbsorbed = rTotalSize + bonus + buffer
        local rLay3Absorbed  = rTotalSize + buffer
        local rLay2Absorbed  = rTotalSize + buffer

        local cur = Chase_State.CurrentLayer

        -- HoldPlayerTrigger
        if cur == 4 then
            if CountInZone(rChaseAbsorbed) > 0 then
                return 4
            else
                return 0
            end
        elseif cur == 3 then
            if CountInZone(rChaseBase) > 0 then return 4 end
            if CountInZone(rLay3Absorbed) > 0 then
                return 3
            else
                return 0
            end
        elseif cur == 2 then
            if CountInZone(rChaseBase) > 0 then return 4 end
            if CountInZone(rLay3Base) > 0 then return 3 end
            if CountInZone(rLay2Absorbed) > 0 then
                return 2
            else
                return 0
            end
        elseif cur == 1 then
            if CountInZone(rChaseBase) > 0 then return 4 end
            if CountInZone(rLay3Base) > 0 then return 3 end
            if CountInZone(rLay2Base) > 0 then return 2 end
            if CountInZone(rLay1Base + buffer) > 0 then
                return 1
            else
                return 0
            end
        else
            if CountInZone(rChaseBase) > 0 then return 4
            elseif CountInZone(rLay3Base) > 0 then return 3
            elseif CountInZone(rLay2Base) > 0 then return 2
            elseif CountInZone(rLay1Base) > 0 then return 1
            else return 0 end
        end
    end

    ApplyTargetLayer = function(targetLayer)
        local inLobby = IsInLobby()
        if inLobby then
            if Chase_State.VoiceA or Chase_State.VoiceB or Chase_State.ThemesSilenced then
                StopChaseAudio(true)
            end
            return
        end

        if not IsLocalKiller() then
            if Chase_State.VoiceA or Chase_State.VoiceB or Chase_State.ThemesSilenced then
                StopChaseAudio(true)
            end
            return
        end

        if IsInCutscene() then
            if Chase_State.VoiceA or Chase_State.VoiceB then
                StopChaseAudio(false)
            end
            return
        end

        if not GetValue("Chase_Active") or (LMS_State and (LMS_State.LMSActive or LMS_State.IsTriggered)) then
            if Chase_State.VoiceA or Chase_State.VoiceB or Chase_State.ThemesSilenced then
                StopChaseAudio(false)
            end
            return
        end

        local trackName = GetValue("Chase_Track") or "None"
        if trackName == "Off" or trackName == "None" then
            if Chase_State.VoiceA or Chase_State.VoiceB or Chase_State.ThemesSilenced then
                StopChaseAudio(false)
            end
            return
        end

        local startDelay = tonumber(GetValue("Chase_StartDelay")) or 2.0
        if (os.clock() - RoundSpawnTime) < startDelay then
            if Chase_State.VoiceA or Chase_State.VoiceB then StopChaseAudio(false) end
            return
        end

        if targetLayer <= 0 then
            if not Chase_State.ZeroSince then
                Chase_State.ZeroSince = os.clock()
            elseif os.clock() - Chase_State.ZeroSince >= 1.0 then
                if Chase_State.VoiceA or Chase_State.VoiceB or Chase_State.ThemesSilenced then
                    StopChaseAudio(false)
                end
            end
            return
        else
            Chase_State.ZeroSince = nil
        end

        SilenceThemesForCustomChase()
        Chase_State.ThemesSilenced = true

        if targetLayer == Chase_State.LastProcessedLayer and trackName == Chase_State.LastProcessedTrack then
            local activeV = (Chase_State.ActiveVoice == "A") and Chase_State.VoiceA or Chase_State.VoiceB
            if activeV and activeV.IsPlaying then
                local baseVol = tonumber(GetValue("Chase_Volume")) or 1.0
                local distFactor = math.clamp(1 - (Chase_State.ClosestDistance / 100), 0, 1)
                activeV.Volume = baseVol * (0.8 + 0.25 * distFactor)
                return
            end
        end

        if not TrackRegistry.Chase[trackName] then
            StopChaseAudio(false)
            return
        end

        local trackData = TrackRegistry.Chase[trackName]
        local assetId = ResolveTrack("Chase", trackName)
        if not assetId then
            StopChaseAudio(false)
            return
        end

        local timings = trackData.Timings
        local hasTimings = (timings.Lay1 or timings.Lay2 or timings.Lay3 or timings.Chase)

        if not hasTimings then
            if targetLayer == 4 then
                if Chase_State.CurrentTierName ~= "ChaseOnly" then
                    PlayChaseSegment(assetId, "ChaseOnly", nil, false, nil, nil)
                end
            else
                StopChaseAudio(false)
            end
            Chase_State.LastProcessedLayer = targetLayer
            Chase_State.LastProcessedTrack = trackName
            Chase_State.CurrentLayer = targetLayer
            return
        end

        local targetTierName = nil
        local targetSegment = nil

        if targetLayer == 4 and timings.Chase then
            targetTierName = "Chase"
            targetSegment = timings.Chase
        elseif targetLayer == 3 and timings.Lay3 then
            targetTierName = "Lay3"
            targetSegment = timings.Lay3
        elseif targetLayer == 2 and timings.Lay2 then
            targetTierName = "Lay2"
            targetSegment = timings.Lay2
        elseif targetLayer == 1 and timings.Lay1 then
            targetTierName = "Lay1"
            targetSegment = timings.Lay1
        end

        if not targetTierName then
            StopChaseAudio(false)
            return
        end

        if Chase_State.IsTransitioning then return end

        if Chase_State.CurrentTierName ~= targetTierName then
            local prevTier = Chase_State.CurrentTierName
            if prevTier == "Lay1" and targetTierName == "Lay2" and timings.Trans1_2 then
                PlayChaseSegment(assetId, "Trans1_2", timings.Trans1_2, true, "Lay2", timings.Lay2)
            elseif prevTier == "Lay2" and targetTierName == "Lay3" and timings.Trans2_3 then
                PlayChaseSegment(assetId, "Trans2_3", timings.Trans2_3, true, "Lay3", timings.Lay3)
            elseif prevTier == "Lay3" and targetTierName == "Chase" and timings.Trans3_Chase then
                PlayChaseSegment(assetId, "Trans3_Chase", timings.Trans3_Chase, true, "Chase", timings.Chase)
            elseif targetTierName == "Chase" and (prevTier == "Lay1" or prevTier == "Lay2") and timings.Trans3_Chase then
                PlayChaseSegment(assetId, "Trans3_Chase", timings.Trans3_Chase, true, "Chase", timings.Chase)
            else
                PlayChaseSegment(assetId, targetTierName, targetSegment, false, nil, nil)
            end
        end

        Chase_State.LastProcessedLayer = targetLayer
        Chase_State.LastProcessedTrack = trackName
        Chase_State.CurrentLayer = targetLayer
    end

    local function HookGameChaseEvents()
        local mod = GetChaseModule()
        if mod and mod.LayerChanged and typeof(mod.LayerChanged.Connect) == "function" then
            mod.LayerChanged:Connect(function(layer)
                if IsLocalKiller() and not IsInLobby() then
                    ApplyTargetLayer(layer or 0)
                end
            end)
        end
    end
    HookGameChaseEvents()

    local function SyncChaseAudio()
        local inLobbyNow = IsInLobby()
        if WasInLobbyState and not inLobbyNow then
            RoundSpawnTime = os.clock()
        end
        WasInLobbyState = inLobbyNow

        if not IsLocalKiller() or inLobbyNow then
            if Chase_State.VoiceA or Chase_State.VoiceB or Chase_State.ThemesSilenced then
                StopChaseAudio(true)
            end
            return
        end

        local activeLayer = ResolveGameActiveLayer()
        ApplyTargetLayer(activeLayer)
    end

    local function HookDropdownExpansion(frame)
        local options = frame:FindFirstChild("Options", true)
        if not options or options:GetAttribute("FPlus_ExpandHook") then return end
        options:SetAttribute("FPlus_ExpandHook", true)

        local row = frame
        while row and row.Parent and row.Parent ~= ContentsScrolling do
            row = row.Parent
        end
        if not (row and row:IsA("GuiObject")) then return end

        local defaultRowZ = row.LayoutOrder > 0 and (50000 - (row.LayoutOrder * 2)) or row.ZIndex
        local defaultFrameZ = frame.ZIndex
        local defaultOptionsZ = options.ZIndex

        options:GetPropertyChangedSignal("Visible"):Connect(function()
            if options.Visible then
                row.ZIndex = 60000
                frame.ZIndex = 60001
                options.ZIndex = 60002

                task.wait(0.04)
                if not options.Visible then return end

                local optBottom = options.AbsolutePosition.Y + options.AbsoluteSize.Y
                local contBottom = ContentsScrolling.AbsolutePosition.Y + ContentsScrolling.AbsoluteSize.Y
                local overflow = optBottom - contBottom

                if overflow > 0 then
                    local extraSpace = overflow + 35
                    TweenService:Create(CanvasAutoSpacer, TweenInfo.new(0.2, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
                        Size = UDim2.new(1, 0, 0, extraSpace)
                    }):Play()

                    local targetScrollY = ContentsScrolling.CanvasPosition.Y + overflow + 25
                    TweenService:Create(ContentsScrolling, TweenInfo.new(0.25, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
                        CanvasPosition = Vector2.new(0, targetScrollY)
                    }):Play()
                else
                    CanvasAutoSpacer.Size = UDim2.new(1, 0, 0, 0)
                end
            else
                row.ZIndex = defaultRowZ
                frame.ZIndex = defaultFrameZ
                options.ZIndex = defaultOptionsZ

                TweenService:Create(CanvasAutoSpacer, TweenInfo.new(0.18, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
                    Size = UDim2.new(1, 0, 0, 0)
                }):Play()
            end
        end)
    end

                    else
                        if SavedOriginalVisibility[item] == nil then
                            SavedOriginalVisibility[item] = item.Visible

            end
        end

    end

    local IsRailOpen = false

    local function OpenRailAnimation()
        if IsRailOpen then return end
        IsRailOpen = true
        TabRail.Visible = true

        for _, data in ipairs(TABS) do
            local ref = ButtonRefs[data.Key]
            if ref then
                ref.Item.Position = UDim2.new(0.5, 45, 0, 0)
                local targetOffset = (ref.Index - (TOTAL_COUNT + 1) / 2) * STEP
                TweenService:Create(ref.Item, CONE_OUT, { Position = UDim2.new(0.5, 0, 0, targetOffset) }):Play()
                TweenService:Create(ref.Scale, CONE_OUT, { Scale = 1 }):Play()
                AnimateButtonFade(ref, true)
            end
        end

        SwitchActiveTab(CurrentTab)
    end

    local function CloseRailAnimation()
        if not IsRailOpen then return end
        IsRailOpen = false

        for _, data in ipairs(TABS) do
            local ref = ButtonRefs[data.Key]
            if ref then
                TweenService:Create(ref.Item, CONE_IN, { Position = UDim2.new(0.5, 45, 0, 0) }):Play()
                TweenService:Create(ref.Scale, CONE_IN, { Scale = 0.05 }):Play()
                AnimateButtonFade(ref, false)
            end
        end

        task.delay(0.23, function()
            if not IsRailOpen then TabRail.Visible = false end
        end)
    end

    local function SyncRailState()
        local isShown = PlusMenu.Visible and (PlusMenu:GetAttribute("Visible") ~= false)
        if isShown and not IsRailOpen then
            OpenRailAnimation()
        elseif not isShown and IsRailOpen then
            CloseRailAnimation()
        end
    end

    PlusMenu:GetPropertyChangedSignal("Visible"):Connect(function()
        SyncRailState()
    end)

    task.delay(0.3, function()
        for _, item in ipairs(ContentsScrolling:GetChildren()) do
            if item:IsA("GuiObject") and not item:IsA("UIListLayout") and not item:IsA("UIPadding") and item ~= CanvasAutoSpacer then
                local ownerTab = FastGetItemTab(item)
                local isAddonItem = (ownerTab == "Custom_LMS" or ownerTab == "Custom_LMS_Visuals" or ownerTab == "Custom_Chase" or ownerTab == "Fun_Wheel" or ownerTab == "Fun_Settings" or ownerTab == "Fun_Credits")
                if not isAddonItem and SavedOriginalVisibility[item] == nil then
                    SavedOriginalVisibility[item] = item.Visible
                end
            end
        end
        if PlusMenu.Visible and (PlusMenu:GetAttribute("Visible") ~= false) then
            OpenRailAnimation()
        end
    end)
