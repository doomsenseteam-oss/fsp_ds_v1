    local function FadeLMSMuteBus(targetVol, duration)
        if LMS_MuteTween then LMS_MuteTween:Cancel(); LMS_MuteTween = nil end
        local dur = math.clamp(duration or 0.5, 0.05, 3.0)
        LMS_MuteTween = TweenService:Create(LMS_MuteBus, TweenInfo.new(dur, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), { Volume = targetVol })
        LMS_MuteTween:Play()
    end

    local function FadeChaseMuteBus(targetVol, duration)
        if Chase_MuteTween then Chase_MuteTween:Cancel(); Chase_MuteTween = nil end
        local dur = math.clamp(duration or 0.6, 0.05, 3.0)
        Chase_MuteTween = TweenService:Create(Chase_MuteBus, TweenInfo.new(dur, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), { Volume = targetVol })
        Chase_MuteTween:Play()
    end

    local function IsLMSSound(snd)
        if not snd or not snd:IsA("Sound") or snd.Name:find("Custom_") then return false end
        local n = snd.Name:lower()
        return (n == "lastsurvivor") or (n:find("lastsurvivor") ~= nil) or (n:find("lms") ~= nil)
    end

    local function SuppressLMSAudio(snd)
        if not snd or not snd:IsA("Sound") or snd.Name:find("Custom_") then return end
        if not SuppressedLMS[snd] then
            SuppressedLMS[snd] = { OriginalGroup = snd.SoundGroup }
            snd.SoundGroup = LMS_MuteBus
        else
            if snd.SoundGroup ~= LMS_MuteBus then snd.SoundGroup = LMS_MuteBus end
        end
    end

    local function RestoreNativeLMS(duration)
        local fadeDur = duration or 0.5
        FadeLMSMuteBus(1, fadeDur)
        task.delay(fadeDur + 0.05, function()
            if not (LMS_State and LMS_State.LMSActive) then
                for snd, data in pairs(SuppressedLMS) do
                    if snd and snd.Parent then
                        pcall(function() snd.SoundGroup = data.OriginalGroup end)
                    end
                end
                table.clear(SuppressedLMS)
            end
        end)
    end

    local function SuppressChaseAudio(snd)
        if not snd or not snd:IsA("Sound") or snd.Name:find("Custom_") then return end
        local nameLow = snd.Name:lower()
        if nameLow:find("lobby") or IsLMSSound(snd) then return end

        if not SuppressedChase[snd] then
            SuppressedChase[snd] = { OriginalGroup = snd.SoundGroup }
            snd.SoundGroup = Chase_MuteBus
        else
            if snd.SoundGroup ~= Chase_MuteBus then snd.SoundGroup = Chase_MuteBus end
        end
    end

        local char = LocalPlayer.Character
        if char then
            char:SetAttribute("MuteChaseThemes", nil)

        FadeChaseMuteBus(1, fadeDur)

        task.delay(fadeDur + 0.05, function()
            if not (Chase_State and Chase_State.ThemesSilenced) then
                for snd, data in pairs(SuppressedChase) do
                    if snd and snd.Parent then
                        pcall(function() snd.SoundGroup = data.OriginalGroup end)
                    end
                end
                table.clear(SuppressedChase)
            end
        end)
    end

    local function IsSpectating()
        local map = workspace:FindFirstChild("Map")
        if not map or not map:FindFirstChild("Ingame") then return false end

        local char = LocalPlayer.Character
        local specFolder = workspace.Players:FindFirstChild("Spectating")
        if specFolder and char and char:IsDescendantOf(specFolder) then
            return true
        end
        if char and char.Parent and char.Parent.Name:lower():find("spectat") then
            return true
        end

        local cam = workspace.CurrentCamera
        local subj = cam and cam.CameraSubject
        local target = subj and (subj:IsA("Humanoid") and subj.Parent or subj:IsA("Model") and subj)
        if target and target:IsA("Model") and target ~= char then
            local survFolder = workspace.Players:FindFirstChild("Survivors")
            local killFolder = workspace.Players:FindFirstChild("Killers")
            if (survFolder and target:IsDescendantOf(survFolder)) or (killFolder and target:IsDescendantOf(killFolder)) then
                return true
            end
        end

        return false
    end

    local function IsInLobby()
        local map = workspace:FindFirstChild("Map")
        if not map or not map:FindFirstChild("Ingame") then return true end

        if IsSpectating() then return false end

        local themes = workspace:FindFirstChild("Themes")
        if themes and themes:FindFirstChild("Lobby") and themes.Lobby:IsA("Sound") and themes.Lobby.IsPlaying then
            return true
        end

        local sideBar = MainUI:FindFirstChild("Sidebar")
        if sideBar and sideBar:GetAttribute("WasVisible") == true then return true end

        return false
    end

    local function IsLocalKiller()
        local char = LocalPlayer.Character
        if not char or char:GetAttribute("Dead") == true then return false end
        local hum = char:FindFirstChildOfClass("Humanoid")
        if not hum or hum.Health <= 0 then return false end
        local killersFolder = workspace.Players:FindFirstChild("Killers")
        return killersFolder and char:IsDescendantOf(killersFolder)
    end

        if IsInLobby() then return false end

        if IsLocalKiller() then return true end
        if GetValue("Chase_SurvivorMode") == true then
            local killFolder = workspace.Players:FindFirstChild("Killers")
            if killFolder and #killFolder:GetChildren() > 0 then
                return true
            end
        end
        return false
    end

    local function SafeList(dir)
        local results = {}
        if not listfiles then return results end
        for _, path in ipairs({ dir, dir:gsub("/", "\\"), dir .. "/", dir:gsub("/", "\\") .. "\\" }) do
            local s, files = pcall(listfiles, path)
            if s and type(files) == "table" and #files > 0 then
                for _, f in ipairs(files) do
                    local fullPath = tostring(f):gsub("\\", "/")
                    if not fullPath:find("/") then
                        fullPath = dir:gsub("\\", "/") .. "/" .. fullPath
                    end
                    results[#results + 1] = fullPath
                end
                break
            end
        end
        return results
    end

    local function ParseChaseTiming(txtPath)
        local timings = {}
        if isfile and readfile then
            local exists = false
            pcall(function() exists = isfile(txtPath) or isfile(txtPath:gsub("/", "\\")) end)
            if exists then
                local content = ""
                pcall(function() content = readfile(isfile(txtPath) and txtPath or txtPath:gsub("/", "\\")) end)
                for line in content:gmatch("[^\r\n]+") do
                    local key, sStart, sEnd = line:match("([%w_%-]+)%s*:%s*([%d%.]+)%s*-%s*([%d%.]+)")
                    if key and sStart and sEnd then
                        local k = key:lower():gsub("%s+", "")
                        local startT, endT = tonumber(sStart), tonumber(sEnd)
                        if startT and endT and endT > startT then
                            if k:find("lay1%-2") or k:find("lay1_2") or k:find("lay1to2") or k:find("layer1%-2") then
                                timings.Trans1_2 = { Start = startT, End = endT }
                            elseif k:find("lay2%-3") or k:find("lay2_3") or k:find("lay2to3") or k:find("layer2%-3") then
                                timings.Trans2_3 = { Start = startT, End = endT }
                            elseif k:find("lay3%-chase") or k:find("lay3_chase") or k:find("lay3tochase") or k:find("layer3%-chase") then
                                timings.Trans3_Chase = { Start = startT, End = endT }
                            elseif k:find("lay1") or k:find("layer1") then
                                timings.Lay1 = { Start = startT, End = endT }
                            elseif k:find("lay2") or k:find("layer2") then
                                timings.Lay2 = { Start = startT, End = endT }
                            elseif k:find("lay3") or k:find("layer3") then
                                timings.Lay3 = { Start = startT, End = endT }
                            elseif k:find("chase") then
                                timings.Chase = { Start = startT, End = endT }
                            end
                        end
                    end
                end
            end
        end
        return timings
    end

    local function ScanAllTracks()
        table.clear(TrackRegistry.Survivor)
        table.clear(TrackRegistry.Killer)
        table.clear(TrackRegistry.Chase)

        local survList = {"None", "Off"}
        local killList = {"None", "Off"}
        local chaseList = {"None", "Off"}
        local seenSurv, seenKill, seenChase = {}, {}, {}

        local folders = { SURV_DIR, KILL_DIR, LMS_DIR, CHASE_DIR, MUSIC_DIR, BASE_ROOT }

        local function CheckFile(fullPath)
            local norm = fullPath:gsub("\\", "/")
            local filename = norm:match("[^/]+$")
            if not filename then return end

            local ext = filename:match("%.([^%.]+)$")
            if ext then
                ext = ext:lower()
                if ext == "mp3" or ext == "ogg" or ext == "wav" then
                    local cleanName = filename:sub(1, #filename - #ext - 1)
                    local lowPath = norm:lower()
                    local lowName = cleanName:lower()

                    if lowPath:find("chase") then
                        if not seenChase[lowName] then
                            seenChase[lowName] = true
                            local txtPath = norm:sub(1, #norm - #ext) .. "txt"
                            TrackRegistry.Chase[cleanName] = {
                                AudioPath = norm,
                                Timings = ParseChaseTiming(txtPath)
                            }
                            table.insert(chaseList, cleanName)
                        end
                    end

                    local isKiller = lowPath:find("/killer") or lowPath:find("\\killer")
                    local isChase  = lowPath:find("chase")
                    if not isKiller and not isChase then
                        if not seenSurv[lowName] then
                            seenSurv[lowName] = true
                            TrackRegistry.Survivor[cleanName] = norm
                            table.insert(survList, cleanName)
                        end
                    end

                    local isSurv = lowPath:find("/survivor") or lowPath:find("\\survivor")
                    if not isSurv and not isChase then
                        if not seenKill[lowName] then
                            seenKill[lowName] = true
                            TrackRegistry.Killer[cleanName] = norm
                            table.insert(killList, cleanName)
                        end
                    end
                end
            end
        end

        for _, fld in ipairs(folders) do
            for _, f in ipairs(SafeList(fld)) do
                CheckFile(f)
            end
        end

        local function SortOptions(tbl)
            table.sort(tbl, function(a, b)
                if a == "None" then return true elseif b == "None" then return false end
                if a == "Off" then return true elseif b == "Off" then return false end
                return a:lower() < b:lower()
            end)
        end
        SortOptions(survList)
        SortOptions(killList)
        SortOptions(chaseList)

        return table.concat(survList, "|"), table.concat(killList, "|"), table.concat(chaseList, "|")
    end

    local function ResolveTrack(role, selectedTrack)
        if not selectedTrack or selectedTrack == "None" or selectedTrack == "Off" then return nil end
        local finalPath = nil
        if role == "Chase" then
            finalPath = TrackRegistry.Chase[selectedTrack] and TrackRegistry.Chase[selectedTrack].AudioPath
        else
            finalPath = TrackRegistry[role] and TrackRegistry[role][selectedTrack]
        end

        if not finalPath then
            ScanAllTracks()
            if role == "Chase" then
                finalPath = TrackRegistry.Chase[selectedTrack] and TrackRegistry.Chase[selectedTrack].AudioPath
            else
                finalPath = TrackRegistry[role] and TrackRegistry[role][selectedTrack]
            end
        end

        if not finalPath and isfile then
            local checkDirs = { SURV_DIR, KILL_DIR, LMS_DIR, CHASE_DIR, MUSIC_DIR, BASE_ROOT }
            for _, d in ipairs(checkDirs) do
                for _, ext in ipairs({ ".mp3", ".ogg", ".wav" }) do
                    local directPath = d .. "/" .. selectedTrack .. ext
                    local exists = false
                    pcall(function() exists = isfile(directPath) or isfile(directPath:gsub("/", "\\")) end)
                    if exists then
                        finalPath = directPath
                        break
                    end
                end
                if finalPath then break end
            end
        end

        if finalPath and (getcustomasset or getsynasset) then
            local func = getcustomasset or getsynasset
            local s, asset = pcall(func, finalPath)
            if s and asset then return asset end
            local s2, asset2 = pcall(func, finalPath:gsub("/", "\\"))
            if s2 and asset2 then return asset2 end
        end
        return nil
    end

    local function DetectRole(model)
        if not model or not model:IsA("Model") then return "Survivor" end
        local killersFolder = workspace.Players:FindFirstChild("Killers")
        local survFolder = workspace.Players:FindFirstChild("Survivors")

        if killersFolder and model:IsDescendantOf(killersFolder) then return "Killer" end
        if survFolder and model:IsDescendantOf(survFolder) then return "Survivor" end

        local hum = model:FindFirstChildOfClass("Humanoid")
        if hum and hum.MaxHealth >= 500 then return "Killer" end

        local clean = model.Name:lower():gsub("[%s_%-]+", "")
        for k, _ in pairs(KILLER_KEYS) do
            if clean:find(k) then return "Killer" end
        end
        return "Survivor"
    end

    local function ResolveEntity(model, fallbackRole)
        local rawNames = {}
        if model and model:IsA("Model") then
            local plr = Players:GetPlayerFromCharacter(model)
            local ActorsMod = ReplicatedStorage:FindFirstChild("Modules") and ReplicatedStorage.Modules:FindFirstChild("Gameplay") and ReplicatedStorage.Modules.Gameplay:FindFirstChild("Actors")
            if plr and ActorsMod then
                pcall(function()
                    local act = require(ActorsMod).CurrentActors[plr]
                    if act and act.Config and act.Config.DisplayName then
                        table.insert(rawNames, act.Config.DisplayName)
                    end
                end)
            end

            table.insert(rawNames, model:GetAttribute("Character"))
            table.insert(rawNames, model:GetAttribute("CharName"))
            table.insert(rawNames, model:GetAttribute("DisplayName"))
            table.insert(rawNames, model:GetAttribute("Actor"))
            table.insert(rawNames, model.Name)
        end

        local sortedKeys = {}
        for k in pairs(CHAR_DATA) do table.insert(sortedKeys, k) end
        table.sort(sortedKeys, function(a, b) return #a > #b end)

        for _, raw in ipairs(rawNames) do
            if raw and typeof(raw) == "string" and raw ~= "" then
                local clean = raw:lower():gsub("[%s_%-]+", "")
                for _, key in ipairs(sortedKeys) do
                    local keyClean = key:lower():gsub("[%s_%-]+", "")
                    if clean == keyClean then
                        return "rbxthumb://type=Asset&id=" .. CHAR_DATA[key].Asset .. "&w=420&h=420", CHAR_DATA[key].Name
                    end
                end
            end
        end

        for _, raw in ipairs(rawNames) do
            if raw and typeof(raw) == "string" and raw ~= "" then
                local clean = raw:lower():gsub("[%s_%-]+", "")
                for _, key in ipairs(sortedKeys) do
                    local keyClean = key:lower():gsub("[%s_%-]+", "")
                    if clean:find(keyClean, 1, true) then
                        return "rbxthumb://type=Asset&id=" .. CHAR_DATA[key].Asset .. "&w=420&h=420", CHAR_DATA[key].Name
                    end
                end
            end
        end

        if fallbackRole == "Killer" then
            return "rbxthumb://type=Asset&id=89705807157114&w=420&h=420", "SLASHER"
        else
            return "rbxthumb://type=Asset&id=131391655751675&w=420&h=420", "NOOB"
        end
    end

        LMSActive = false,
        VisualPlayed = false,

    local function StopLMS(instant)
        LMS_State.IsTriggered = false
        LMS_State.LMSActive = false
        LMS_State.VisualPlayed = false
        LMS_State.StopAt = nil
        LMS_State.CurrentTrackName = nil

        if FX_Gui then
            pcall(function() FX_Gui:ClearAllChildren() end)
        end

        local fadeDur = instant and 0 or 0.5

        if LMS_State.CurrentSound then
            local snd = LMS_State.CurrentSound
            LMS_State.CurrentSound = nil
            if instant or fadeDur <= 0.05 then
                snd:Stop()
                snd:Destroy()
            else
                local tween = TweenService:Create(snd, TweenInfo.new(fadeDur, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), { Volume = 0 })
                tween:Play()
                tween.Completed:Once(function()
                    snd:Stop()
                    snd:Destroy()
                end)
                Debris:AddItem(snd, fadeDur + 0.2)
            end
        end

        RestoreNativeLMS(fadeDur)
    end

    local function PlayLocalLMS(assetId, trackName, preserveTime)
        local prevSound = LMS_State.CurrentSound
        local prevTime = 0
        if prevSound and prevSound.IsPlaying then
            prevTime = prevSound.TimePosition
            local fadeOut = TweenService:Create(prevSound, TweenInfo.new(0.5, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), { Volume = 0 })
            fadeOut:Play()
            fadeOut.Completed:Once(function()
                prevSound:Stop()
                prevSound:Destroy()
            end)
            Debris:AddItem(prevSound, 0.7)
            LMS_State.CurrentSound = nil
        elseif prevSound then
            prevSound:Stop()
            prevSound:Destroy()
            LMS_State.CurrentSound = nil
        end

        local vol = tonumber(GetValue("LMS_Volume")) or 1.0
        local loop = (GetValue("LMS_Loop") == true)

        local snd = Instance.new("Sound")
        snd.Name = "Custom_LMS_Audio"
        snd.SoundId = assetId
        snd.Volume = 0
        snd.Looped = loop
        snd.PlaybackSpeed = 1.0
        snd.Parent = SoundService

        local function StartAudio()
            if not snd or not snd.Parent then return end
            if preserveTime and prevTime > 0 then
                if snd.TimeLength > 0 then
                    snd.TimePosition = prevTime % snd.TimeLength
                else
                    snd.TimePosition = prevTime
                end
            else
                snd.TimePosition = 0
            end
            snd:Play()
            TweenService:Create(snd, TweenInfo.new(0.5, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), { Volume = vol }):Play()
        end

        if not snd.IsLoaded then
            snd.Loaded:Once(StartAudio)
        else
            StartAudio()
        end

        LMS_State.CurrentSound = snd
        LMS_State.CurrentTrackName = trackName
        LMS_Cache.LastPlayedAsset = assetId
        LMS_Cache.LastPlayedTrackName = trackName
    end

    local function TriggerLMSVisualEffect()
        if LMS_State.VisualPlayed then return end
        local fxType = GetValue("LMS_StartEffect") or "None"
        local fxFunc = StartVisuals[fxType]
        if typeof(fxFunc) == "function" then
            LMS_State.VisualPlayed = true
            task.spawn(fxFunc, false)
        end
    end

    local function CheckTrueLMS()
        if not GetValue("LMS_Active") then
            if LMS_State.IsTriggered or LMS_State.CurrentSound or LMS_State.LMSActive then
                StopLMS(false)
            end
            return
        end

        local inLobby = IsInLobby()
        local allowLobbyAudio = (GetValue("LMS_LobbyAudio") == true)

        local survivorsFolder = workspace.Players:FindFirstChild("Survivors")
        local spectatingFolder = workspace.Players:FindFirstChild("Spectating")
        local killersFolder = workspace.Players:FindFirstChild("Killers")

        if not survivorsFolder or not killersFolder then
            if LMS_State.IsTriggered or LMS_State.LMSActive then StopLMS(false) end
            return
        end

        local myCharacter = LocalPlayer.Character
        local myHum = myCharacter and myCharacter:FindFirstChildOfClass("Humanoid")
        local isAlive = myCharacter and (myCharacter:GetAttribute("Dead") ~= true) and (myHum and myHum.Health > 0)

        local isLocalSurvivor = isAlive and myCharacter:IsDescendantOf(survivorsFolder)
        local isLocalKiller = isAlive and myCharacter:IsDescendantOf(killersFolder)
        local isNonParticipant = not isLocalSurvivor and not isLocalKiller

        if isNonParticipant and not allowLobbyAudio then
            if LMS_State.IsTriggered or LMS_State.CurrentSound or LMS_State.LMSActive then
                StopLMS(false)
            end
            return
        end

        local themes = workspace:FindFirstChild("Themes")
        local isNativeLMS = false
        local activeNativeSounds = {}

        if themes then
            for _, snd in ipairs(themes:GetChildren()) do
                if snd:IsA("Sound") and IsLMSSound(snd) then
                    table.insert(activeNativeSounds, snd)
                    if (snd.IsPlaying or snd.Playing or snd.TimePosition > 0) then
                        isNativeLMS = true
                    end
                end
            end
        end

        local activeSurvivors = 0
        local totalSurvivors = 0

        for _, char in ipairs(survivorsFolder:GetChildren()) do
            totalSurvivors = totalSurvivors + 1
            local hum = char:FindFirstChildOfClass("Humanoid")
            local isDead = (char:GetAttribute("Dead") == true) or (hum and hum.Health <= 0)
            local isSpectating = spectatingFolder and char:IsDescendantOf(spectatingFolder)

            if not isDead and not isSpectating then
                activeSurvivors = activeSurvivors + 1
            end
        end

        local isLMS = false
        if IsCurrentMatch2v12 then
            if isNativeLMS and activeSurvivors <= 3 and activeSurvivors >= 1 then
                isLMS = true
            elseif activeSurvivors <= 3 and activeSurvivors >= 1 and totalSurvivors > 3 and not inLobby then
                isLMS = true
            end
        else
            if isNativeLMS and activeSurvivors == 1 then
                isLMS = true
            elseif activeSurvivors == 1 and totalSurvivors > 1 and not inLobby then
                isLMS = true
            end
        end

        if isNonParticipant and not allowLobbyAudio then
            isLMS = false
        end

        if isLMS then
            LMS_State.StopAt = nil

            local role = "Survivor"
            local selectedTrack = "None"

            if IsCurrentMatch2v12 then
                -- 2vs3
                if isLocalKiller then
                    role = "Killer"
                    selectedTrack = GetValue("TwoVThree_KillerTrack") or "None"
                    if selectedTrack == "None" then
                        selectedTrack = GetValue("LMS_KillerTrack") or "None"
                    end
                else
                    role = "Survivor"
                    selectedTrack = GetValue("TwoVThree_SurvTrack") or "None"
                    if selectedTrack == "None" then
                        selectedTrack = GetValue("LMS_SurvTrack") or "None"
                    end
                end
            else
                if isLocalKiller then
                    role = "Killer"
                    selectedTrack = GetValue("LMS_KillerTrack") or "None"
                else
                    role = "Survivor"
                    selectedTrack = GetValue("LMS_SurvTrack") or "None"
                end
            end

            if selectedTrack == "Off" then
                if LMS_State.CurrentTrackName ~= "Off" or LMS_State.LMSActive then
                    LMS_State.CurrentTrackName = "Off"
                    LMS_State.LMSActive = false
                    if LMS_State.CurrentSound then
                        local s = LMS_State.CurrentSound
                        LMS_State.CurrentSound = nil
                        local tw = TweenService:Create(s, TweenInfo.new(0.5, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), { Volume = 0 })
                        tw:Play()
                        tw.Completed:Once(function() s:Stop(); s:Destroy() end)
                        Debris:AddItem(s, 0.7)
                    end
                    RestoreNativeLMS(0.5)
                end
                return
            end

            if selectedTrack == "None" then
                if LMS_State.CurrentTrackName ~= "None" then
                    LMS_State.CurrentTrackName = "None"
                    LMS_State.LMSActive = true
                    if LMS_State.CurrentSound then
                        local s = LMS_State.CurrentSound
                        LMS_State.CurrentSound = nil
                        local tw = TweenService:Create(s, TweenInfo.new(0.5, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), { Volume = 0 })
                        tw:Play()
                        tw.Completed:Once(function() s:Stop(); s:Destroy() end)
                        Debris:AddItem(s, 0.7)
                    end
                end
                LMS_State.LMSActive = true
                for _, s in ipairs(activeNativeSounds) do
                    SuppressLMSAudio(s)
                end
                FadeLMSMuteBus(0, 0.4)
                return
            end

            LMS_State.LMSActive = true
            for _, s in ipairs(activeNativeSounds) do
                SuppressLMSAudio(s)
            end
            FadeLMSMuteBus(0, 0.4)

            -- Hotswap
            if LMS_State.IsTriggered and LMS_State.CurrentTrackName ~= selectedTrack then
                LMS_State.CurrentTrackName = selectedTrack
                local assetId = ResolveTrack(role, selectedTrack)
                if assetId then
                    PlayLocalLMS(assetId, selectedTrack, true)
                end
            end

            if not LMS_State.IsTriggered then
                LMS_State.IsTriggered = true
                local delaySec = tonumber(GetValue("LMS_StartDelay")) or 0

                task.delay(delaySec, function()
                    if not LMS_State.IsTriggered or not GetValue("LMS_Active") then return end
                    if IsInLobby() and not (GetValue("LMS_LobbyAudio") == true) then return end

                    TriggerLMSVisualEffect()

                    if selectedTrack ~= "None" and selectedTrack ~= "Off" then
                        local assetId = ResolveTrack(role, selectedTrack)
                        if assetId then PlayLocalLMS(assetId, selectedTrack, false) end
                    end
                end)
            else
                if LMS_State.CurrentSound and not LMS_State.CurrentSound.IsPlaying then
                    LMS_State.CurrentSound:Play()
                end
            end
        else
            if LMS_State.IsTriggered or LMS_State.LMSActive then
                if not LMS_State.StopAt then
                    LMS_State.StopAt = os.clock() + 1.5
                elseif os.clock() >= LMS_State.StopAt then
                    StopLMS(false)
                end
            end
        end
    end

    local function IsInCutscene()
        local themes = workspace:FindFirstChild("Themes")
        if themes then
            for _, snd in ipairs(themes:GetChildren()) do
                if snd:IsA("Sound") and (snd.IsPlaying or snd.Playing) then
                    local n = snd.Name:lower()
                    if n:find("victory") or n:find("intro") or n:find("execution") then
                        return true
                    end
                end
            end
        end

        local char = LocalPlayer.Character
        if char then
            if char:GetAttribute("InCutscene") == true
                or char:GetAttribute("Executing") == true
                or char:GetAttribute("InExecution") == true
                or char:GetAttribute("IntroActive") == true then
                return true
            end

            local hum = char:FindFirstChildOfClass("Humanoid")
            local anim = hum and hum:FindFirstChildOfClass("Animator")
            if anim then
                for _, track in ipairs(anim:GetPlayingAnimationTracks()) do
                    local tName = (track.Name or ""):lower()
                    if tName:find("execution") or tName:find("intro") or tName:find("victory") then
                        return true
                    end
                end
            end
        end

        local rTimer = ReplicatedStorage:FindFirstChild("RoundTimer")
        if rTimer and (rTimer.Value == "Round begins in:" or rTimer.Value == "Waiting for more players.") then
            return true
        end

        return false
    end

    local function LoadTrackConfig(trackName)
        if not isfile or not readfile or not trackName or trackName == "None" or trackName == "Off" then return end
        local cfgPath = GetTrackConfigPath(trackName)
        if not cfgPath then return end

        local norm = cfgPath:gsub("/", "\\")
        local exists = false
        pcall(function() exists = isfile(cfgPath) or isfile(norm) end)

        local trackData = TrackRegistry.Chase[trackName]
        local timings = trackData and trackData.Timings or {}

        if not exists then
            IsConfigLoading = true
            local function ApplyVal(key, val)
                local inst = GetValue(key, true)
                if inst and val ~= nil then inst.Value = val end
            end
            ApplyVal("Chase_HasLay1", timings.Lay1 ~= nil)
            ApplyVal("Chase_HasLay2", timings.Lay2 ~= nil)
            ApplyVal("Chase_HasLay3", timings.Lay3 ~= nil)
            ApplyVal("Chase_HasChase", true)
            task.delay(0.1, function()
                IsConfigLoading = false
                SaveTrackConfig(trackName)
            end)
            return
        end

        local rawContent = ""
        pcall(function()
            rawContent = readfile(isfile(cfgPath) and cfgPath or norm)
        end)

        local s, data = pcall(function() return HttpService:JSONDecode(rawContent) end)
        if s and type(data) == "table" then
            IsConfigLoading = true
            local function ApplyVal(key, val)
                local inst = GetValue(key, true)
                if inst and val ~= nil then
                    inst.Value = val
                end
            end

            ApplyVal("Chase_Volume", data.Volume)
            ApplyVal("Chase_FadeTime", data.FadeTime)
            ApplyVal("Chase_StartDelay", data.StartDelay)
            ApplyVal("Chase_RangeMultiplier", data.RangeMultiplier)
            ApplyVal("Chase_OverlapBuffer", data.OverlapBuffer)
            ApplyVal("Chase_BonusExpansion", data.BonusExpansion)
            ApplyVal("Chase_DistChase", data.DistChase)
            ApplyVal("Chase_DistLay3", data.DistLay3)
            ApplyVal("Chase_DistLay2", data.DistLay2)
            ApplyVal("Chase_DistLay1", data.DistLay1)
            ApplyVal("Chase_PlaybackSpeed", data.PlaybackSpeed)

            if data.HasLay1 ~= nil then ApplyVal("Chase_HasLay1", data.HasLay1) else ApplyVal("Chase_HasLay1", timings.Lay1 ~= nil) end
            if data.HasLay2 ~= nil then ApplyVal("Chase_HasLay2", data.HasLay2) else ApplyVal("Chase_HasLay2", timings.Lay2 ~= nil) end
            if data.HasLay3 ~= nil then ApplyVal("Chase_HasLay3", data.HasLay3) else ApplyVal("Chase_HasLay3", timings.Lay3 ~= nil) end
            if data.HasChase ~= nil then ApplyVal("Chase_HasChase", data.HasChase) else ApplyVal("Chase_HasChase", true) end

            task.delay(0.1, function() IsConfigLoading = false end)
        end
    end

            char:SetAttribute("MuteChaseThemes", true)
        end

        local themes = workspace:FindFirstChild("Themes")
        if not themes then return end

        for _, snd in ipairs(themes:GetChildren()) do
            if snd:IsA("Sound") and not IsSoundProtected(snd) and not IsLMSSound(snd) then
                SuppressChaseAudio(snd)
            end
        end

        local dur = fadeTime or (tonumber(GetValue("Chase_FadeTime")) or 0.6)
        FadeChaseMuteBus(0, dur)
    end

    local function StopChaseAudio(instant, keepNativeMuted)
        Chase_State.CurrentLayer = 0
        Chase_State.CurrentTierName = nil
        Chase_State.IsTransitioning = false
        Chase_State.IsLoading = false
        Chase_State.ZeroSince = nil
        Chase_State.LastProcessedLayer = -1
        Chase_State.LastProcessedTrack = nil

        if Chase_State.LoopConnA then Chase_State.LoopConnA:Disconnect(); Chase_State.LoopConnA = nil end
        if Chase_State.LoopConnB then Chase_State.LoopConnB:Disconnect(); Chase_State.LoopConnB = nil end

        local fadeTime = instant and 0 or (tonumber(GetValue("Chase_FadeTime")) or 0.6)

        local function KillVoice(v)
            if not v or not v.Parent then return end
            if instant or fadeTime <= 0.05 then
                v:Stop()
                v:Destroy()
            else
                if v:GetAttribute("FadingOut") then return end
                v:SetAttribute("FadingOut", true)

                local tw = TweenService:Create(v, TweenInfo.new(fadeTime, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), { Volume = 0 })
                tw:Play()
                tw.Completed:Once(function()
                    v:Stop()
                    v:Destroy()
                end)
                Debris:AddItem(v, fadeTime + 0.25)
            end
        end

        KillVoice(Chase_State.VoiceA); Chase_State.VoiceA = nil
        KillVoice(Chase_State.VoiceB); Chase_State.VoiceB = nil
        Chase_State.ActiveAsset = nil

        for _, snd in ipairs(SoundService:GetChildren()) do
            if snd:IsA("Sound") and snd.Name:find("Custom_Chase_Voice") then
                KillVoice(snd)
            end
        end

        if not keepNativeMuted and Chase_State.ThemesSilenced then
            Chase_State.ThemesSilenced = false
            RestoreNativeChase(fadeTime)
        end
    end

    -- [ Dual-Voice Crossfader ]
    local function PlayChaseSegment(assetId, tierName, segment, isTransition, nextTierName, nextSegment)
        Chase_State.CurrentTierName = tierName
        Chase_State.IsLoading = true
        Chase_State.IsTransitioning = isTransition == true

        local baseVol = tonumber(GetValue("Chase_Volume")) or 1.0
        local pitch = tonumber(GetValue("Chase_PlaybackSpeed")) or 1.0
        local fadeTime = math.clamp(tonumber(GetValue("Chase_FadeTime")) or 0.6, 0.1, 3.0)
        local distFactor = math.clamp(1 - (Chase_State.ClosestDistance / 100), 0, 1)
        local targetVol = baseVol * (0.8 + 0.25 * distFactor)

        local currentVoice = (Chase_State.ActiveVoice == "A") and Chase_State.VoiceA or Chase_State.VoiceB
        local targetVoiceSlot = (Chase_State.ActiveVoice == "A") and "B" or "A"

        local prevSlotVoice = (targetVoiceSlot == "B") and Chase_State.VoiceB or Chase_State.VoiceA
        if prevSlotVoice and prevSlotVoice ~= currentVoice and not prevSlotVoice:GetAttribute("FadingOut") then
            prevSlotVoice:Stop()
            prevSlotVoice:Destroy()
        end

        local newVoice = Instance.new("Sound")
        newVoice.Name = "Custom_Chase_Voice_" .. targetVoiceSlot
        newVoice.SoundId = assetId
        newVoice.Volume = 0
        newVoice.PlaybackSpeed = pitch
        newVoice.Parent = SoundService

        if targetVoiceSlot == "B" then
            if Chase_State.LoopConnB then Chase_State.LoopConnB:Disconnect(); Chase_State.LoopConnB = nil end
            Chase_State.VoiceB = newVoice
        else
            if Chase_State.LoopConnA then Chase_State.LoopConnA:Disconnect(); Chase_State.LoopConnA = nil end
            Chase_State.VoiceA = newVoice
        end

        Chase_State.ActiveVoice = targetVoiceSlot
        Chase_State.ActiveAsset = assetId

        if currentVoice and currentVoice.IsPlaying and not currentVoice:GetAttribute("FadingOut") then
            currentVoice:SetAttribute("FadingOut", true)
            local fadeOut = TweenService:Create(currentVoice, TweenInfo.new(fadeTime, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), { Volume = 0 })
            fadeOut:Play()
            fadeOut.Completed:Once(function()
                currentVoice:Stop()
                currentVoice:Destroy()
            end)
            Debris:AddItem(currentVoice, fadeTime + 0.25)
        end

        local function SetupLoop()
            Chase_State.IsLoading = false
            if not newVoice or not newVoice.Parent then return end
            if segment and segment.Start and segment.End then
                newVoice.Looped = false
                newVoice.TimePosition = segment.Start

                if isTransition then
                    local conn
                    conn = RunService.Heartbeat:Connect(function()
                        if newVoice and newVoice.IsPlaying then
                            if newVoice.TimePosition >= segment.End then
                                Chase_State.IsTransitioning = false
                                if conn then conn:Disconnect() end
                                if nextTierName and nextSegment then
                                    PlayChaseSegment(assetId, nextTierName, nextSegment, false, nil, nil)
                                end
                            end
                        else
                            if conn then conn:Disconnect() end
                        end
                    end)
                    if targetVoiceSlot == "B" then Chase_State.LoopConnB = conn else Chase_State.LoopConnA = conn end
                else
                    local conn
                    conn = RunService.Heartbeat:Connect(function()
                        if newVoice and newVoice.IsPlaying then
                            if newVoice.TimePosition >= segment.End then
                                newVoice.TimePosition = segment.Start
                            end
                        else
                            if conn then conn:Disconnect() end
                        end
                    end)
                    if targetVoiceSlot == "B" then Chase_State.LoopConnB = conn else Chase_State.LoopConnA = conn end
                end
            else
                newVoice.Looped = true
            end

            newVoice:Play()
            local fadeIn = TweenService:Create(newVoice, TweenInfo.new(fadeTime, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), { Volume = targetVol })
            fadeIn:Play()
        end

        if not newVoice.IsLoaded then
            newVoice.Loaded:Once(SetupLoop)
        else
            SetupLoop()
        end
    end

    local function ResolveGameActiveLayer()
        if IsInLobby() then return 0 end
        if not CanRunChase() then return 0 end
        if IsInCutscene() then return 0 end

        local hasLay1 = GetValue("Chase_HasLay1") == true
        local hasLay2 = GetValue("Chase_HasLay2") == true
        local hasLay3 = GetValue("Chase_HasLay3") == true
        local hasChase = GetValue("Chase_HasChase") ~= false

        local mult    = math.max(tonumber(GetValue("Chase_RangeMultiplier")) or 1.0, 0.1)
        local buffer  = math.clamp(tonumber(GetValue("Chase_OverlapBuffer")) or 3.57, 0, 15)
        local bonus   = math.max(tonumber(GetValue("Chase_BonusExpansion")) or 25.0, 0)

        local rChaseBase = (tonumber(GetValue("Chase_DistChase")) or 22) * mult
        local rLay3Base  = (tonumber(GetValue("Chase_DistLay3")) or 40) * mult
        local rLay2Base  = (tonumber(GetValue("Chase_DistLay2")) or 60) * mult
        local rLay1Base  = (tonumber(GetValue("Chase_DistLay1")) or 85) * mult

        local rTotalSize = (hasLay1 and rLay1Base)
            or (hasLay2 and rLay2Base)
            or (hasLay3 and rLay3Base)
            or rChaseBase

        local rChaseAbsorbed = rTotalSize + bonus + buffer
        local rLay3Absorbed  = (hasLay1 or hasLay2) and (rTotalSize + buffer) or (rLay3Base + buffer)
        local rLay2Absorbed  = hasLay1 and (rTotalSize + buffer) or (rLay2Base + buffer)

        local cur = Chase_State.CurrentLayer
        local survDistances = {}
        local minD = 9999

        if IsLocalKiller() then
            local myChar = LocalPlayer.Character
            if not myChar or myChar:GetAttribute("Dead") == true then return 0 end
            local myRoot = myChar:FindFirstChild("HumanoidRootPart") or myChar.PrimaryPart
            local survFolder = workspace.Players:FindFirstChild("Survivors")
            if not myRoot or not survFolder then return 0 end

            for _, sChar in ipairs(survFolder:GetChildren()) do
                local sHum = sChar:FindFirstChildOfClass("Humanoid")
                local sRoot = sChar:FindFirstChild("HumanoidRootPart") or sChar.PrimaryPart
                if sRoot and (not sHum or sHum.Health > 0) and sChar:GetAttribute("Dead") ~= true then
                    local d = (sRoot.Position - myRoot.Position).Magnitude
                    table.insert(survDistances, d)
                    if d < minD then minD = d end
                end
            end
        else
            local killFolder = workspace.Players:FindFirstChild("Killers")
            local survFolder = workspace.Players:FindFirstChild("Survivors")
            if not killFolder or not survFolder then return 0 end

            local activeKillers = {}
            for _, kChar in ipairs(killFolder:GetChildren()) do
                local kHum = kChar:FindFirstChildOfClass("Humanoid")
                local kRoot = kChar:FindFirstChild("HumanoidRootPart") or kChar.PrimaryPart
                if kRoot and (not kHum or kHum.Health > 0) and kChar:GetAttribute("Dead") ~= true then
                    table.insert(activeKillers, kRoot)
                end
            end
            if #activeKillers == 0 then return 0 end

            local targetRoot = nil
            local myChar = LocalPlayer.Character
            local myHum = myChar and myChar:FindFirstChildOfClass("Humanoid")
            local isAliveSurvivor = myChar and (myChar:GetAttribute("Dead") ~= true) and (myHum and myHum.Health > 0) and myChar:IsDescendantOf(survFolder)

            if isAliveSurvivor then
                targetRoot = myChar:FindFirstChild("HumanoidRootPart") or myChar.PrimaryPart
            else
                local cam = workspace.CurrentCamera
                local subj = cam and cam.CameraSubject
                local spectatedChar = subj and (subj:IsA("Humanoid") and subj.Parent or subj:IsA("Model") and subj)

                if spectatedChar and spectatedChar:IsA("Model") then
                    if spectatedChar:IsDescendantOf(killFolder) then
                        local kRoot = spectatedChar:FindFirstChild("HumanoidRootPart") or spectatedChar.PrimaryPart
                        if kRoot then
                            for _, sChar in ipairs(survFolder:GetChildren()) do
                                local sHum = sChar:FindFirstChildOfClass("Humanoid")
                                local sRoot = sChar:FindFirstChild("HumanoidRootPart") or sChar.PrimaryPart
                                if sRoot and (not sHum or sHum.Health > 0) and sChar:GetAttribute("Dead") ~= true then
                                    local d = (sRoot.Position - kRoot.Position).Magnitude
                                    table.insert(survDistances, d)
                                    if d < minD then minD = d end
                                end
                            end
                        end
                    else
                        targetRoot = spectatedChar:FindFirstChild("HumanoidRootPart") or spectatedChar.PrimaryPart
                    end
                end
            end

            if targetRoot then
                for _, kRoot in ipairs(activeKillers) do
                    local d = (kRoot.Position - targetRoot.Position).Magnitude
                    if d < minD then minD = d end
                end
                table.insert(survDistances, minD)
            end
        end

        Chase_State.ClosestDistance = minD

        local function CountInZone(radius)
            local c = 0
            for _, d in ipairs(survDistances) do
                if d <= radius then c = c + 1 end
            end
            return c
        end

        -- HoldPlayerTrigger
        if cur == 4 then
            if hasChase and CountInZone(rChaseAbsorbed) > 0 then
                return 4
            else
                return 0
            end
        elseif cur == 3 then
            if hasChase and CountInZone(rChaseBase) > 0 then return 4 end
            if hasLay3 and CountInZone(rLay3Absorbed) > 0 then
                return 3
            else
                return 0
            end
        elseif cur == 2 then
            if hasChase and CountInZone(rChaseBase) > 0 then return 4 end
            if hasLay3 and CountInZone(rLay3Base) > 0 then return 3 end
            if hasLay2 and CountInZone(rLay2Absorbed) > 0 then
                return 2
            else
                return 0
            end
        elseif cur == 1 then
            if hasChase and CountInZone(rChaseBase) > 0 then return 4 end
            if hasLay3 and CountInZone(rLay3Base) > 0 then return 3 end
            if hasLay2 and CountInZone(rLay2Base) > 0 then return 2 end
            if hasLay1 and CountInZone(rLay1Base + buffer) > 0 then
                return 1
            else
                return 0
            end
        else
            if hasChase and CountInZone(rChaseBase) > 0 then return 4
            elseif hasLay3 and CountInZone(rLay3Base) > 0 then return 3
            elseif hasLay2 and CountInZone(rLay2Base) > 0 then return 2
            elseif hasLay1 and CountInZone(rLay1Base) > 0 then return 1
            else return 0 end
        end
    end

    ApplyTargetLayer = function(targetLayer)
        local inLobby = IsInLobby()
        if inLobby then
            if Chase_State.VoiceA or Chase_State.VoiceB or Chase_State.ThemesSilenced then
                StopChaseAudio(true)
            end
            return
        end

        if not CanRunChase() then
            if Chase_State.VoiceA or Chase_State.VoiceB or Chase_State.ThemesSilenced then
                StopChaseAudio(true)
            end
            return
        end

        if IsInCutscene() then
            if Chase_State.VoiceA or Chase_State.VoiceB then
                StopChaseAudio(false)
            end
            return
        end

        if not GetValue("Chase_Active") or (LMS_State and (LMS_State.LMSActive or LMS_State.IsTriggered)) then
            if Chase_State.VoiceA or Chase_State.VoiceB or Chase_State.ThemesSilenced then
                StopChaseAudio(false)
            end
            return
        end

        local trackName = GetValue("Chase_Track") or "None"
        if trackName == "Off" or trackName == "None" then
            if Chase_State.VoiceA or Chase_State.VoiceB or Chase_State.ThemesSilenced then
                StopChaseAudio(false)
            end
            return
        end

        local startDelay = tonumber(GetValue("Chase_StartDelay")) or 2.0
        if (os.clock() - RoundSpawnTime) < startDelay then
            if Chase_State.VoiceA or Chase_State.VoiceB then StopChaseAudio(false) end
            return
        end

        if targetLayer <= 0 then
            if not Chase_State.ZeroSince then
                Chase_State.ZeroSince = os.clock()
            elseif os.clock() - Chase_State.ZeroSince >= 1.0 then
                if Chase_State.VoiceA or Chase_State.VoiceB or Chase_State.ThemesSilenced then
                    StopChaseAudio(false)
                end
            end
            return
        else
            Chase_State.ZeroSince = nil
        end

        local fadeDuration = tonumber(GetValue("Chase_FadeTime")) or 0.6
        if not Chase_State.ThemesSilenced then
            SilenceThemesForCustomChase(fadeDuration)
            Chase_State.ThemesSilenced = true
        end

        local trackData = TrackRegistry.Chase[trackName]
        local assetId = ResolveTrack("Chase", trackName)
        local isTrackHotswap = (Chase_State.LastProcessedTrack ~= trackName) or (Chase_State.ActiveAsset ~= assetId)

        if targetLayer == Chase_State.LastProcessedLayer and not isTrackHotswap then
            local activeV = (Chase_State.ActiveVoice == "A") and Chase_State.VoiceA or Chase_State.VoiceB
            if activeV and activeV.IsPlaying then
                local baseVol = tonumber(GetValue("Chase_Volume")) or 1.0
                local distFactor = math.clamp(1 - (Chase_State.ClosestDistance / 100), 0, 1)
                activeV.Volume = baseVol * (0.8 + 0.25 * distFactor)
                return
            end
        end

        if not trackData or not assetId then
            StopChaseAudio(false, true)
            return
        end

        local timings = trackData.Timings
        local hasTimings = (timings.Lay1 or timings.Lay2 or timings.Lay3 or timings.Chase)

        if not hasTimings then
            if targetLayer == 4 then
                SilenceThemesForCustomChase(fadeDuration)
                if Chase_State.CurrentTierName ~= "ChaseOnly" or isTrackHotswap then
                    PlayChaseSegment(assetId, "ChaseOnly", nil, false, nil, nil)
                end
            else
                StopChaseAudio(false, true)
            end
            Chase_State.LastProcessedLayer = targetLayer
            Chase_State.LastProcessedTrack = trackName
            Chase_State.CurrentLayer = targetLayer
            return
        end

        local hasLay1 = GetValue("Chase_HasLay1") == true
        local hasLay2 = GetValue("Chase_HasLay2") == true
        local hasLay3 = GetValue("Chase_HasLay3") == true
        local hasChase = GetValue("Chase_HasChase") ~= false

        local targetTierName = nil
        local targetSegment = nil

        if targetLayer == 4 and hasChase and timings.Chase then
            targetTierName = "Chase"
            targetSegment = timings.Chase
        elseif targetLayer == 3 and hasLay3 and timings.Lay3 then
            targetTierName = "Lay3"
            targetSegment = timings.Lay3
        elseif targetLayer == 2 and hasLay2 and timings.Lay2 then
            targetTierName = "Lay2"
            targetSegment = timings.Lay2
        elseif targetLayer == 1 and hasLay1 and timings.Lay1 then
            targetTierName = "Lay1"
            targetSegment = timings.Lay1
        end

        if not targetTierName then
            if Chase_State.VoiceA or Chase_State.VoiceB or Chase_State.CurrentTierName ~= nil then
                StopChaseAudio(false, true)
            end
            Chase_State.LastProcessedLayer = targetLayer
            Chase_State.LastProcessedTrack = trackName
            Chase_State.CurrentLayer = targetLayer
            return
        end

        if (Chase_State.IsTransitioning or Chase_State.IsLoading) and not isTrackHotswap then return end

        if Chase_State.CurrentTierName ~= targetTierName or isTrackHotswap then
            local prevTier = Chase_State.CurrentTierName
            if isTrackHotswap then
                PlayChaseSegment(assetId, targetTierName, targetSegment, false, nil, nil)
            elseif prevTier == "Lay1" and targetTierName == "Lay2" and timings.Trans1_2 then
                PlayChaseSegment(assetId, "Trans1_2", timings.Trans1_2, true, "Lay2", timings.Lay2)
            elseif prevTier == "Lay2" and targetTierName == "Lay3" and timings.Trans2_3 then
                PlayChaseSegment(assetId, "Trans2_3", timings.Trans2_3, true, "Lay3", timings.Lay3)
            elseif prevTier == "Lay3" and targetTierName == "Chase" and timings.Trans3_Chase then
                PlayChaseSegment(assetId, "Trans3_Chase", timings.Trans3_Chase, true, "Chase", timings.Chase)
            elseif targetTierName == "Chase" and (prevTier == "Lay1" or prevTier == "Lay2") and timings.Trans3_Chase then
                PlayChaseSegment(assetId, "Trans3_Chase", timings.Trans3_Chase, true, "Chase", timings.Chase)
            else
                PlayChaseSegment(assetId, targetTierName, targetSegment, false, nil, nil)
            end
        end

        Chase_State.LastProcessedLayer = targetLayer
        Chase_State.LastProcessedTrack = trackName
        Chase_State.CurrentLayer = targetLayer
    end

    local function HookGameChaseEvents()
        local mod = GetChaseModule()
        if mod and mod.LayerChanged and typeof(mod.LayerChanged.Connect) == "function" then
            mod.LayerChanged:Connect(function(layer)
                if not CanRunChase() or IsInLobby() then return end
                if not GetValue("Chase_Active") then return end

                local rawLayer = tonumber(layer) or 0
                if rawLayer > 0 then
                    if not Chase_State.ThemesSilenced then
                        SilenceThemesForCustomChase()
                        Chase_State.ThemesSilenced = true
                    end
                else
                    if Chase_State.CurrentLayer <= 0 and not (LMS_State and LMS_State.LMSActive) then
                        Chase_State.ThemesSilenced = false
                        RestoreNativeChase()
                    end
                end
            end)
        end
    end
    HookGameChaseEvents()

    local function SyncChaseAudio()
        local inLobbyNow = IsInLobby()
        if WasInLobbyState and not inLobbyNow then
            RoundSpawnTime = os.clock()
        end
        WasInLobbyState = inLobbyNow

        if not CanRunChase() or inLobbyNow then
            if Chase_State.VoiceA or Chase_State.VoiceB or Chase_State.ThemesSilenced then
                StopChaseAudio(true)
            end
            return
        end

        local activeLayer = ResolveGameActiveLayer()
        ApplyTargetLayer(activeLayer)

        local isChaseRunning = GetValue("Chase_Active") and (Chase_State and Chase_State.CurrentLayer > 0)
        local themes = workspace:FindFirstChild("Themes")

        if isChaseRunning then
            if themes then
                for _, snd in ipairs(themes:GetChildren()) do
                    if snd:IsA("Sound") and not IsSoundProtected(snd) and not IsLMSSound(snd) then
                        if snd.SoundGroup ~= Chase_MuteBus then
                            SuppressChaseAudio(snd)
                        end
                    end
                end
            end
            FadeChaseMuteBus(0, 0.2)
        else
            if Chase_State and not Chase_State.ThemesSilenced and next(SuppressedChase) ~= nil then
                RestoreNativeChase(0.4)
            end
        end
    end

    local function RestoreNightWorldLights()
        for light, orig in pairs(NightLightCache) do
            if light and light.Parent then
                pcall(function()
                    light.Color = orig.Color
                    light.Brightness = orig.Brightness
                    light.Range = orig.Range
                end)
            end
        end
        table.clear(NightLightCache)
    end

    local function UpdateNightWorldLights()
        if not GetValue("Night_Enabled") then
            RestoreNightWorldLights()
            return
        end
        local inLobby = IsInLobby()
        if inLobby and not GetValue("Night_Lobby") then
            RestoreNightWorldLights()
            return
        end

        local modLights = GetValue("Night_ModifyLights")
        if not modLights then
            RestoreNightWorldLights()
            return
        end

        local brightVal = tonumber(GetValue("Night_LightBright")) or 2.0
        local rangeVal = tonumber(GetValue("Night_LightRange")) or 25
        local colorKey = GetValue("Night_LightColorPreset") or "Pure White"
        local targetColor = NightColorPalettes.Lights[colorKey] or Color3.fromRGB(255, 255, 255)

        for _, light in ipairs(workspace:GetDescendants()) do
            if light:IsA("PointLight") or light:IsA("SpotLight") or light:IsA("SurfaceLight") then
                if not NightLightCache[light] then
                    NightLightCache[light] = {
                        Color = light.Color,
                        Brightness = light.Brightness,
                        Range = light.Range
                    }
                end
                light.Color = targetColor
                light.Brightness = brightVal
                light.Range = rangeVal
            end
        end
    end

    local function RestoreNightLighting()
        if not NightOriginal.Saved then return end
        IsNightApplyingInternal = true
        pcall(function()
            if NightOriginal.TimeOfDay then Lighting.TimeOfDay = NightOriginal.TimeOfDay end
            if NightOriginal.Ambient then Lighting.Ambient = NightOriginal.Ambient end
            if NightOriginal.OutdoorAmbient then Lighting.OutdoorAmbient = NightOriginal.OutdoorAmbient end
            if NightOriginal.Brightness then Lighting.Brightness = NightOriginal.Brightness end
            if NightOriginal.FogColor then Lighting.FogColor = NightOriginal.FogColor end
            if NightOriginal.FogStart then Lighting.FogStart = NightOriginal.FogStart end
            if NightOriginal.FogEnd then Lighting.FogEnd = NightOriginal.FogEnd end

            local customAtmos = Lighting:FindFirstChild("Doomsense_NightAtmosphere")
            if customAtmos then
                customAtmos:Destroy()
            end

            local atmos = Lighting:FindFirstChildOfClass("Atmosphere")
            if atmos and NightOriginal.Atmosphere then
                atmos.Density = NightOriginal.Atmosphere.Density
                atmos.Color = NightOriginal.Atmosphere.Color
                atmos.Decay = NightOriginal.Atmosphere.Decay
                atmos.Offset = NightOriginal.Atmosphere.Offset
                atmos.Glare = NightOriginal.Atmosphere.Glare
                atmos.Haze = NightOriginal.Atmosphere.Haze
            end
        end)
        RestoreNightWorldLights()
        NightOriginal.Saved = false
        IsNightApplyingInternal = false
    end

    local function ApplyNightLighting()
        local enabled = GetValue("Night_Enabled")
        local inLobby = IsInLobby()
        local allowInLobby = GetValue("Night_Lobby")

        if not enabled or (inLobby and not allowInLobby) then
            RestoreNightLighting()
            return
        end

        SaveOriginalLighting()
        IsNightApplyingInternal = true

        local timeChoice = GetValue("Night_TimePreset") or "Midnight (00:00)"
        local timeString = NightColorPalettes.Time[timeChoice] or "00:00:00"

        local brightness = tonumber(GetValue("Night_Brightness")) or 1.5

        local ambChoice = GetValue("Night_AmbientPreset") or "Navy Night"
        local ambientColor = NightColorPalettes.Ambient[ambChoice] or Color3.fromRGB(50, 50, 80)

        local outChoice = GetValue("Night_OutdoorPreset") or "Pitch Black"
        local outdoorColor = NightColorPalettes.Outdoor[outChoice] or Color3.fromRGB(0, 0, 0)

        local fogEnabled = GetValue("Night_FogEnabled")
        local fogChoice = GetValue("Night_FogPreset") or "Navy Blue"
        local fogColor = NightColorPalettes.Fog[fogChoice] or Color3.fromRGB(50, 50, 80)
        local fogDensity = tonumber(GetValue("Night_FogDensity")) or 2.5
        local fogDistance = tonumber(GetValue("Night_FogDistance")) or 350
        local hazeVal = tonumber(GetValue("Night_AtmosphereHaze")) or 1.0

        pcall(function()
            Lighting.TimeOfDay = timeString
            Lighting.Ambient = ambientColor
            Lighting.OutdoorAmbient = outdoorColor
            Lighting.Brightness = brightness

            local atmos = Lighting:FindFirstChildOfClass("Atmosphere")
            if fogEnabled then
                Lighting.FogColor = fogColor
                Lighting.FogStart = 0
                Lighting.FogEnd   = fogDistance

                if not atmos then
                    atmos = Instance.new("Atmosphere")
                    atmos.Name = "Doomsense_NightAtmosphere"
                    atmos.Parent = Lighting
                    NightOriginal.AtmosphereCreated = true
                end
                atmos.Density = math.clamp(fogDensity / 5, 0, 1)
                atmos.Color   = fogColor
                atmos.Decay   = fogColor
                atmos.Offset  = 0.25
                atmos.Glare   = 0
                atmos.Haze    = hazeVal
            else
                Lighting.FogStart = 0
                Lighting.FogEnd   = 9e9
                if atmos then
                    if NightOriginal.AtmosphereCreated then
                        atmos:Destroy()
                        NightOriginal.AtmosphereCreated = false
                    elseif NightOriginal.Atmosphere then
                        atmos.Density = NightOriginal.Atmosphere.Density
                        atmos.Color   = NightOriginal.Atmosphere.Color
                        atmos.Haze    = NightOriginal.Atmosphere.Haze
                    else
                        atmos.Density = 0
                    end
                end
            end
        end)

        UpdateNightWorldLights()
        IsNightApplyingInternal = false
    end

    local WasNightLobbyState = true
    RunService.Heartbeat:Connect(function()
        local inLobby = IsInLobby()
        if inLobby ~= WasNightLobbyState then
            WasNightLobbyState = inLobby
            if inLobby then
                if not GetValue("Night_Lobby") then
                    RestoreNightLighting()
                else
                    if GetValue("Night_Enabled") then
                        task.delay(0.2, ApplyNightLighting)
                    end
                end
            else
                if GetValue("Night_Enabled") then
                    task.delay(0.6, ApplyNightLighting)
                end
            end
        end
    end)

    Lighting:GetPropertyChangedSignal("TimeOfDay"):Connect(function()
        if not IsNightApplyingInternal and GetValue("Night_Enabled") then
            local inLobby = IsInLobby()
            if not inLobby or GetValue("Night_Lobby") then
                ApplyNightLighting()
            end
        end
    end)

    if InGame then
        InGame.ChildAdded:Connect(function(Child)
            if Child.Name == "Map" and GetValue("Night_Enabled") then
                NightOriginal.Saved = false
                task.delay(1.2, ApplyNightLighting)
            end
        end)
    end

    local function ClearAllGenBars()

    local function UpdateGeneratorWorldBars()
        local enabled = GetValue("GenIndicator_Enabled")
        local inLobby = IsInLobby()
        local camera = workspace.CurrentCamera
        local activeMap = GameMap or (workspace:FindFirstChild("Map") and workspace.Map:FindFirstChild("Ingame") and workspace.Map.Ingame:FindFirstChild("Map"))

        if not enabled or inLobby or not activeMap or not camera then
            ClearAllGenBars()
            return
        end

        local gens = activeMap:QueryDescendants("Model#Generator:has(#Main)")
        if #gens == 0 then
            ClearAllGenBars()
            return
        end

        local chosenColorName = GetValue("GeneratorsColor") or "Cyan"
        local barColor = ColorPresets[chosenColorName] or Color3.fromRGB(0, 255, 255)

        local offsetY = tonumber(GetValue("GenIndicator_OffsetY")) or 5.5
        local minDistance = math.clamp(tonumber(GetValue("GenIndicator_MinDist")) or 25, 5, 80)
        local autoHide = GetValue("GenIndicator_AutoHide")
        local genCheck = GetValue("GeneratorsCheck")

        local camPos = camera.CFrame.Position
        local currentGens = {}

        for _, gen in ipairs(gens) do
            local main = gen:FindFirstChild("Main")
            if main then
                currentGens[gen] = true

                local barUi = gen:FindFirstChild("BarUI", true)
                local barPart = barUi and barUi:FindFirstChild("Bar")
                local genProgress = barPart and barPart.Size.X.Scale or (gen:GetAttribute("Completed") and 1 or 0)
                genProgress = math.clamp(genProgress, 0, 1)

                local maxSteps = gen:GetAttribute("MaxCounter")
                if not maxSteps then
                    local maxValObj = gen:FindFirstChild("MaxCounter", true)
                    maxSteps = maxValObj and maxValObj.Value
                end
                maxSteps = tonumber(maxSteps) or 5

                local currentSteps = gen:GetAttribute("Counter")
                if not currentSteps then
                    local curValObj = gen:FindFirstChild("Counter", true)
                    currentSteps = curValObj and curValObj.Value
                end
                currentSteps = tonumber(currentSteps)
                if not currentSteps then
                    if genProgress >= 0.99 or gen:GetAttribute("Completed") == true then
                        currentSteps = maxSteps
                    else
                        currentSteps = math.clamp(math.round(genProgress * maxSteps), 0, maxSteps)
                    end
                end

                local isGenCompleted = (genProgress >= 0.99 or gen:GetAttribute("Completed") == true or currentSteps >= maxSteps)
                local individualCounterText = string.format("%d/%d", currentSteps, maxSteps)
                local individualFillRatio = math.clamp(genProgress, 0, 1)

                local isFinishedHide = (autoHide and isGenCompleted) or (genCheck and isGenCompleted)

                local distToCamera = (camPos - main.Position).Magnitude
                local isTooClose = distToCamera <= minDistance
                local isTooFar = distToCamera >= 1000

                local shouldBeVisible = (not isFinishedHide) and (not isTooClose) and (not isTooFar)

                local data = GenWorldBars[gen]
                if not data or not data.Gui or not data.Gui.Parent then
                    local bGui = Instance.new("BillboardGui")
                    bGui.Name = "FPlus_GenWorldBar"
                    bGui.AlwaysOnTop = true
                    bGui.ResetOnSpawn = false
                    bGui.ClipsDescendants = false
                    bGui.MaxDistance = 1200
                    bGui.Adornee = main

                    bGui.Size = UDim2.new(1.8, 30, 0.45, 8)

                    local bg = Instance.new("Frame")
                    bg.Name = "BG"
                    bg.BorderSizePixel = 1
                    bg.BorderColor3 = Color3.fromRGB(0, 0, 0)
                    bg.BackgroundColor3 = Color3.fromRGB(15, 15, 20)
                    bg.BackgroundTransparency = 1
                    bg.Size = UDim2.fromScale(1, 1)
                    bg.Parent = bGui

                    local fill = Instance.new("Frame")
                    fill.Name = "Fill"
                    fill.BorderSizePixel = 0
                    fill.Size = UDim2.new(individualFillRatio, 0, 1, 0)
                    fill.BackgroundColor3 = barColor
                    fill.BackgroundTransparency = 1
                    fill.Parent = bg

                    local label = Instance.new("TextLabel")
                    label.Name = "CounterLabel"
                    label.BackgroundTransparency = 1
                    label.BorderSizePixel = 0
                    label.AnchorPoint = Vector2.new(0.5, 0.5)
                    label.Position = UDim2.fromScale(0.5, 0.5)
                    label.Size = UDim2.fromScale(1, 1)
                    label.TextXAlignment = Enum.TextXAlignment.Center
                    label.TextYAlignment = Enum.TextYAlignment.Center
                    label.Font = Enum.Font.GothamBold
                    label.TextSize = 13
                    label.TextColor3 = Color3.fromRGB(255, 255, 255)
                    label.TextStrokeTransparency = 1
                    label.TextStrokeColor3 = Color3.fromRGB(0, 0, 0)
                    label.Text = individualCounterText
                    label.ZIndex = 5
                    label.Parent = bg

                    bGui.Parent = main
                    data = {
                        Gui = bGui,
                        Bg = bg,
                        Fill = fill,
                        Label = label,
                        IsVisible = false
                    }
                    GenWorldBars[gen] = data
                end

                data.Gui.StudsOffset = Vector3.new(0, offsetY, 0)

                data.Label.Position = UDim2.fromScale(0.5, 0.5)
                data.Label.Text = individualCounterText

                data.Fill.BackgroundColor3 = barColor
                data.Fill.Size = UDim2.new(individualFillRatio, 0, 1, 0)

                if shouldBeVisible and not data.IsVisible then
                    data.IsVisible = true
                    TweenService:Create(data.Bg, TweenInfo.new(0.15, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), { BackgroundTransparency = 0.3 }):Play()
                    TweenService:Create(data.Fill, TweenInfo.new(0.15, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), { BackgroundTransparency = 0 }):Play()
                    TweenService:Create(data.Label, TweenInfo.new(0.15, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), { TextTransparency = 0, TextStrokeTransparency = 0.35 }):Play()
                elseif not shouldBeVisible and data.IsVisible then
                    data.IsVisible = false
                    TweenService:Create(data.Bg, TweenInfo.new(0.15, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), { BackgroundTransparency = 1 }):Play()
                    TweenService:Create(data.Fill, TweenInfo.new(0.15, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), { BackgroundTransparency = 1 }):Play()
                    TweenService:Create(data.Label, TweenInfo.new(0.15, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), { TextTransparency = 1, TextStrokeTransparency = 1 }):Play()
                end
            end
        end

        for gen, data in pairs(GenWorldBars) do
            if not currentGens[gen] or not gen.Parent then
                if data and data.Gui and data.Gui.Parent then
                    data.Gui:Destroy()
                end
                GenWorldBars[gen] = nil
            end
        end
    end

    local function ClearGraffitiESP()
        for _, data in pairs(GraffitiCache) do

                if data.Highlight and data.Highlight.Parent then data.Highlight:Destroy() end

    local function UpdateVeeGraffitiESP()
        local enabled = GetValue("VeeGraffiti_Enabled")
        local inLobby = IsInLobby()
        local camera = workspace.CurrentCamera
        local camPos = camera and camera.CFrame.Position

        if not enabled or inLobby or not camPos then
            ClearGraffitiESP()
            return
        end

        local colorKey = GetValue("VeeGraffiti_Color") or "Pink-Purple"
        local espColor = ColorPresets[colorKey] or Color3.fromRGB(235, 140, 255)
        local showText = GetValue("VeeGraffiti_ShowText") ~= false
        local maxDist = 1000

        local activeTargets = {}
        local ingameFolder = InGame or (workspace:FindFirstChild("Map") and workspace.Map:FindFirstChild("Ingame"))

        if ingameFolder then
            for _, child in ipairs(ingameFolder:GetChildren()) do
                local isSprayModel = child:IsA("Model") and (child.Name:match("Spray$") or child:FindFirstChild("GraffitiCL"))

                if isSprayModel then
                    local targetHitbox = child:FindFirstChild("Hitbox") or child:FindFirstChildWhichIsA("BasePart")
                    if targetHitbox and targetHitbox:IsA("BasePart") then
                        activeTargets[targetHitbox] = true
                        local dist = (camPos - targetHitbox.Position).Magnitude

                        if dist <= maxDist then
                            local data = GraffitiCache[targetHitbox]
                            if not data or not data.Billboard or not data.Billboard.Parent then
                                local hl = Instance.new("Highlight")
                                hl.Name = HttpService:GenerateGUID(false):sub(1, 7)
                                hl.Adornee = targetHitbox
                                hl.FillColor = espColor
                                hl.OutlineColor = espColor
                                hl.FillTransparency = 0.5
                                hl.OutlineTransparency = 0
                                hl.Parent = targetHitbox

                                local bb = Instance.new("BillboardGui")
                                bb.Name = HttpService:GenerateGUID(false):sub(1, 7)
                                bb.Adornee = targetHitbox
                                bb.AlwaysOnTop = true
                                bb.ResetOnSpawn = true
                                bb.Size = UDim2.new(1.5, 60, 0.75, 25)
                                bb.StudsOffset = Vector3.new(0, 4.2, 0)
                                bb.Parent = targetHitbox

                                local lbl = Instance.new("TextLabel")
                                lbl.Name = "TextLabel"
                                lbl.AnchorPoint = Vector2.new(0.5, 0.5)
                                lbl.Position = UDim2.fromScale(0.5, 0.5)
                                lbl.Size = UDim2.fromScale(1, 1)
                                lbl.BackgroundTransparency = 1
                                lbl.TextScaled = true
                                lbl.TextSize = 8
                                lbl.TextColor3 = espColor
                                lbl.TextStrokeTransparency = 0
                                lbl.TextStrokeColor3 = Color3.fromRGB(0, 0, 0)
                                lbl.Text = "Vee Graffiti"
                                lbl.Parent = bb

                                data = { Highlight = hl, Billboard = bb, Label = lbl }
                                GraffitiCache[targetHitbox] = data
                            end

                            data.Highlight.FillColor = espColor
                            data.Highlight.OutlineColor = espColor
                            data.Highlight.Enabled = true
                            data.Billboard.Enabled = showText
                            data.Label.TextColor3 = espColor
                            data.Label.Text = "Vee Graffiti"

                            if dist <= 8 or dist >= maxDist then
                                data.Label.TextTransparency = 1
                                data.Label.TextStrokeTransparency = 1
                            else
                                data.Label.TextTransparency = 0
                                data.Label.TextStrokeTransparency = 0
                            end
                        else
                            if GraffitiCache[targetHitbox] then
                                if GraffitiCache[targetHitbox].Highlight then GraffitiCache[targetHitbox].Highlight.Enabled = false end
                                if GraffitiCache[targetHitbox].Billboard then GraffitiCache[targetHitbox].Billboard.Enabled = false end
                            end
                        end
                    end
                end
            end
        end

        for targetHitbox, data in pairs(GraffitiCache) do
            if not activeTargets[targetHitbox] or not targetHitbox.Parent then
                if data.Highlight and data.Highlight.Parent then data.Highlight:Destroy() end
                if data.Billboard and data.Billboard.Parent then data.Billboard:Destroy() end
                GraffitiCache[targetHitbox] = nil
            end
        end
    end

    local function HookDropdownExpansion(frame)
        local options = frame:FindFirstChild("Options", true)
        if not options or options:GetAttribute("FPlus_ExpandHook") then return end
        options:SetAttribute("FPlus_ExpandHook", true)

        local row = frame
        while row and row.Parent and row.Parent ~= ContentsScrolling do
            row = row.Parent
        end
        if not (row and row:IsA("GuiObject")) then return end

        local defaultRowZ = row.LayoutOrder > 0 and (50000 - (row.LayoutOrder * 2)) or row.ZIndex
        local defaultFrameZ = frame.ZIndex
        local defaultOptionsZ = options.ZIndex

        options:GetPropertyChangedSignal("Visible"):Connect(function()
            if options.Visible then
                row.ZIndex = 60000
                frame.ZIndex = 60001
                options.ZIndex = 60002

                task.wait(0.04)
                if not options.Visible then return end

                local optBottom = options.AbsolutePosition.Y + options.AbsoluteSize.Y
                local contBottom = ContentsScrolling.AbsolutePosition.Y + ContentsScrolling.AbsoluteSize.Y
                local overflow = optBottom - contBottom

                if overflow > 0 then
                    local extraSpace = overflow + 35
                    TweenService:Create(CanvasAutoSpacer, TweenInfo.new(0.2, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
                        Size = UDim2.new(1, 0, 0, extraSpace)
                    }):Play()

                    local targetScrollY = ContentsScrolling.CanvasPosition.Y + overflow + 25
                    TweenService:Create(ContentsScrolling, TweenInfo.new(0.25, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
                        CanvasPosition = Vector2.new(0, targetScrollY)
                    }):Play()
                else
                    CanvasAutoSpacer.Size = UDim2.new(1, 0, 0, 0)
                end
            else
                row.ZIndex = defaultRowZ
                frame.ZIndex = defaultFrameZ
                options.ZIndex = defaultOptionsZ

                TweenService:Create(CanvasAutoSpacer, TweenInfo.new(0.18, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
                    Size = UDim2.new(1, 0, 0, 0)
                }):Play()
            end
        end)
    end

    local function SwitchActiveTab(tabKey)
        CurrentTab = tabKey
        for key, ref in pairs(ButtonRefs) do
            SetTabVisual(ref, key == tabKey)
        end

        TweenService:Create(CanvasAutoSpacer, TweenInfo.new(0.15, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
            Size = UDim2.new(1, 0, 0, 0)
        }):Play()

        for _, item in ipairs(ContentsScrolling:GetChildren()) do
            if item:IsA("GuiObject") and not item:IsA("UIListLayout") and not item:IsA("UIPadding") and item ~= CanvasAutoSpacer then
                local ownerTab = FastGetItemTab(item)
                local isVisualItem = (ownerTab == "Visual_NightMode" or ownerTab == "Visual_Generators" or ownerTab == "Visual_Camera" or ownerTab == "Visual_VeeGraffiti")
                local isMusicItem = (ownerTab == "Custom_LMS" or ownerTab == "Custom_2v3" or ownerTab == "Custom_LMS_Visuals" or ownerTab == "Custom_Chase")
                local isFunItem = (ownerTab == "Fun_Wheel" or ownerTab == "Fun_Settings" or ownerTab == "Fun_Credits")

                if tabKey == "Main" then
                    if isVisualItem or isMusicItem or isFunItem then
                        item.Visible = false
                    else
                        if SavedOriginalVisibility[item] ~= nil then
                            item.Visible = SavedOriginalVisibility[item]
                        end
                    end
                elseif tabKey == "Visual" then
                    if isVisualItem then
                        item.Visible = true
                    else
                        if SavedOriginalVisibility[item] == nil then
                            SavedOriginalVisibility[item] = item.Visible
                        end
                        item.Visible = false
                    end
                elseif tabKey == "Music" then
                    if isMusicItem then
                        item.Visible = true
                    else
                        if SavedOriginalVisibility[item] == nil then
                            SavedOriginalVisibility[item] = item.Visible
                        end
                        item.Visible = false
                    end
                elseif tabKey == "Fun" then
                    if isFunItem then
                        item.Visible = true
                    else
                        if SavedOriginalVisibility[item] == nil then
                            SavedOriginalVisibility[item] = item.Visible
                        end
                        item.Visible = false
                    end
                end
            end
        end

        ContentsScrolling.CanvasPosition = Vector2.zero
    end

    local function OpenRailAnimation()
        if IsRailOpen then return end
        IsRailOpen = true
        TabRail.Visible = true

        for _, data in ipairs(TABS) do
            local ref = ButtonRefs[data.Key]
            if ref then
                ref.Item.Position = UDim2.new(0.5, 45, 0, 0)
                local targetOffset = (ref.Index - (TOTAL_COUNT + 1) / 2) * STEP
                TweenService:Create(ref.Item, CONE_OUT, { Position = UDim2.new(0.5, 0, 0, targetOffset) }):Play()
                TweenService:Create(ref.Scale, CONE_OUT, { Scale = 1 }):Play()

            end

    end

    local function CloseRailAnimation()
        if not IsRailOpen then return end
        IsRailOpen = false

        for _, data in ipairs(TABS) do
            local ref = ButtonRefs[data.Key]
            if ref then
                TweenService:Create(ref.Item, CONE_IN, { Position = UDim2.new(0.5, 45, 0, 0) }):Play()
                TweenService:Create(ref.Scale, CONE_IN, { Scale = 0.05 }):Play()
                AnimateButtonFade(ref, false)
            end
        end

        task.delay(0.23, function()
            if not IsRailOpen then TabRail.Visible = false end
        end)
    end

    local function SyncRailState()
        local isShown = PlusMenu.Visible and (PlusMenu:GetAttribute("Visible") ~= false)
        if isShown and not IsRailOpen then
            OpenRailAnimation()
        elseif not isShown and IsRailOpen then
            CloseRailAnimation()
        end
    end
